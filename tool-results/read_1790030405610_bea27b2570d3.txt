    62→async function rpcMeta(name:string,args:any={}){
    63→  if(!/^[a-z0-9_]+$/i.test(name))throw new Error('rpc_name_invalid');
    64→  const key=name+':'+Object.keys(args).sort().join(',');
    65→  if(rpcMetaCache.has(key))return rpcMetaCache.get(key);
    66→  const rows=await sql`
    67→    select p.oid::text as oid,p.pronargs,p.pronargdefaults,p.proretset,
    68→      coalesce(p.proargnames[1:p.pronargs],array[]::text[]) as arg_names,
    69→      array(select format_type(t.oid,null)
    70→            from unnest(p.proargtypes::oid[]) with ordinality u(type_oid,ord)
    71→            join pg_type t on t.oid=u.type_oid order by u.ord) as arg_types
    72→    from pg_proc p join pg_namespace n on n.oid=p.pronamespace
    73→    where n.nspname='public' and p.proname=${name}`;
    74→  const supplied=new Set(Object.keys(args));
    75→  const candidates=rows.filter((r:any)=>{
    76→    const names=[...(r.arg_names||[])];
    77→    const required=names.slice(0,Math.max(0,Number(r.pronargs)-Number(r.pronargdefaults||0)));
    78→    return [...supplied].every(k=>names.includes(k))&&required.every((k:string)=>supplied.has(k));
    79→  });
    80→  if(candidates.length!==1)throw new Error(`rpc_signature_ambiguous:${name}:${candidates.length}`);
    81→  rpcMetaCache.set(key,candidates[0]);
    82→  return candidates[0];
    83→}
    84→async function rpc(name:string,args:any={}){
    85→  const m=await rpcMeta(name,args);
    86→  const names=[...m.arg_names];
    87→  const types=[...m.arg_types];
    88→  const params:any[]=[];
    89→  const clauses:string[]=[];
    90→  for(const [k,v] of Object.entries(args)){
    91→    const i=names.indexOf(k);
    92→    if(i<0)throw new Error(`rpc_arg_missing:${name}:${k}`);
    93→    params.push(rpcParam(v,types[i]));
    94→    clauses.push(`\"${k}\" => $${params.length}::${types[i]}`);
    95→  }
    96→  const call=`public.\"${name}\"(${clauses.join(',')})`;
    97→  if(m.proretset){
    98→    const rows=await sql.unsafe(`select to_jsonb(x) as value from ${call} x`,params);
    99→    return rows.map((r:any)=>r.value);
   100→  }
   101→  const rows=await sql.unsafe(`select to_jsonb(${call}) as value`,params);
   102→  return rows[0]?.value??null;
   103→}

   150→// P1-2 multi-writer repair: plane ownership semantics for the persisted state.
   151→// /v1/state accepts three writers (bootstrap heartbeat, 5s supervisor heartbeat,
   152→// realtime observation push). The former full-JSON replacement meant any writer
   153→// that omitted a plane erased it server-side (live-observed as supervisor_mesh
   154→// flapping). The persisted state now merges per top-level plane:
   155→//   - a key present in the incoming payload (even explicit null) overwrites;
   156→//   - a key ABSENT from the incoming payload is preserved.
   157→// boundedState therefore keeps the absent-vs-null distinction: identity and
   158→// scalar fields are always emitted (they are mandatory for every writer), while
