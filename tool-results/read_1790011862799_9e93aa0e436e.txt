     1→import crypto from 'node:crypto';
     2→
     3→export const SUPERVISOR_KEEPALIVE_VERSION = '1.5.0';
     4→export const SUPERVISOR_ID = 'METAENGINE_SUPERVISOR';
     5→export const KEEPALIVE_STATES = Object.freeze([
     6→  'ACTIVE','WAITING','WAKE_PENDING','WAKE_AMBIGUOUS',
     7→  'ROLLOVER_DEFERRED','ROLLOVER_REQUIRED','ROLLOVER_PENDING','ROLLOVER_AMBIGUOUS',
     8→  'PAUSED','RECOVERING','PARKED',
     9→]);
    10→
    11→const AGENT_PLATFORM_CONVERSATION_RE = /^https:\/\/chat\.z\.ai\/c\/[a-z0-9-]+(?:[/?#].*)?$/i;
    12→const WAKE_REASONS = new Set([
    13→  'CONTINUE_DEVELOPMENT',
    14→  'WORKER_RESULT_READY','WORKER_FAILED','WORKER_LOST','CI_TERMINAL',
    15→  'INTEGRATION_HEAD_CHANGED','MILESTONE_READY_FOR_REVIEW',
    16→  'SUPERVISOR_RECOVERY_REQUIRED','WATCHDOG_DEADLINE','RESEARCH_ACCELERATOR_DUE',
    17→]);
    18→const MAX_QUEUED_WAKES = 32;
    19→const MAX_WAKE_HISTORY = 32;
    20→const MAX_WORKER_GENERATION_MEMORY = 2048;
    21→const ADMISSION_STATES = new Set(['UNKNOWN', 'OPEN', 'CLOSED']);
    22→
    23→const clone = (value) => value == null ? value : structuredClone(value);
    24→const iso = (clock) => new Date(clock()).toISOString();
    25→
    26→function normalizeUrl(value) {
    27→  const url = String(value || '').trim();
    28→  if (!AGENT_PLATFORM_CONVERSATION_RE.test(url)) throw new Error('keepalive_supervisor_conversation_invalid');
    29→  return url;
    30→}
    31→
    32→function sanitizeProcessIncarnationId(value) {
    33→  const id = String(value || '').trim();
    34→  return id && id.length <= 160 ? id : null;
    35→}
    36→
    37→function sanitizeActiveWake(input) {
    38→  if (!input || typeof input !== 'object') return null;
    39→  const reason = String(input.reason || '');
    40→  const wakeId = String(input.wake_id || '');
    41→  if (!WAKE_REASONS.has(reason) || !/^wake_[a-z0-9-]+$/i.test(wakeId)) return null;
    42→  return {
    43→    wake_id: wakeId,
    44→    reason,
    45→    queue_key: input.queue_key ? String(input.queue_key).slice(0, 240) : null,
    46→    prepared_at: input.prepared_at || null,
    47→    confirmed_at: input.confirmed_at || null,
    48→    supervisor_epoch: Math.max(1, Number(input.supervisor_epoch) || 1),
    49→    cycle_seq: Math.max(1, Number(input.cycle_seq) || 1),
    50→    process_incarnation_id: sanitizeProcessIncarnationId(input.process_incarnation_id),
    51→    origin_process_incarnation_id: sanitizeProcessIncarnationId(input.origin_process_incarnation_id),
    52→  };
    53→}
    54→
    55→function sanitizeWorkerGenerationMap(input) {
    56→  if (!input || typeof input !== 'object' || Array.isArray(input)) return {};
    57→  return Object.fromEntries(Object.entries(input)
    58→    .filter(([id]) => String(id || '').length > 0)
    59→    .slice(-MAX_WORKER_GENERATION_MEMORY)
    60→    .map(([id, generation]) => [String(id), String(generation || 'UNKNOWN').toUpperCase()]));
    61→}

    63→function freshState() {
    64→  return {
    65→    schema: 'metaengine.supervisor-keepalive.state.v1',
    66→    version: SUPERVISOR_KEEPALIVE_VERSION,
    67→    supervisor_id: SUPERVISOR_ID,
    68→    supervisor_epoch: 1,
    69→    cycle_seq: 0,
    70→    state: 'RECOVERING',
    71→    conversation_url: null,
    72→    tab_id: null,
    73→    paused: false,
    74→    paused_from_state: null,
    75→    process_incarnation_id: null,
    76→    process_incarnation_started_at: null,
    77→    predecessor_process_incarnation_id: null,
    78→    predecessor_fenced_at: null,
    79→    predecessor_queued_wake_count: 0,
    80→    admission_state: 'UNKNOWN',
    81→    admission_reason: 'NOT_OBSERVED',
    82→    admission_generation_floor: null,
    83→    admission_refill_enabled: null,
    84→    admission_supervisor_enabled: null,
    85→    admission_observed_at: null,
    86→    parked_at: null,
    87→    parked_reason: null,
    88→    parked_queued_wake_count: 0,
    89→    parked_wake_reasons: [],
    90→    suppressed_wake_count: 0,
    91→    last_suppressed_wake_at: null,
    92→    last_suppressed_wake_reason: null,
    93→    queued_wakes: [],
    94→    pending_wake: null,
    95→    active_wake: null,
    96→    predecessor_wake_history: [],
    97→    ambiguous_history: [],
    98→    last_wake_at: null,
    99→    last_wake_reason: null,
   100→    // D-K5: timestamp of the last wake that was prepared but provably not
   101→    // sent (pre-effect failure). Rate-limits the retry storm; cleared on a
   102→    // confirmed send.
   103→    last_unsent_attempt_at: null,
   104→    last_completed_cycle_at: null,
   105→    last_research_wake_at: null,
   106→    previous_worker_generation: {},
   107→    rollover_reason: null,
   108→    rollover_release_at: null,
   109→    rollover_attempt: null,
   110→    updated_at: null,
   111→    authority_effect: false,
   112→  };
   113→}
   114→
   115→function sanitize(input) {
   116→  const base = freshState();
   117→  if (!input || input.schema !== base.schema || input.supervisor_id !== SUPERVISOR_ID) return base;
   118→  let conversationUrl = null;
   119→  try { if (input.conversation_url) conversationUrl = normalizeUrl(input.conversation_url); } catch {}
   120→  const queued = Array.isArray(input.queued_wakes)
   121→    ? input.queued_wakes.filter((row) => row && WAKE_REASONS.has(String(row.reason))).slice(-MAX_QUEUED_WAKES).map(clone)
   122→    : [];
   123→  const ambiguousHistory = Array.isArray(input.ambiguous_history)
   124→    ? input.ambiguous_history.filter((row) => row && typeof row === 'object').slice(-MAX_WAKE_HISTORY).map(clone)
   125→    : [];
   126→  const predecessorWakeHistory = Array.isArray(input.predecessor_wake_history)
   127→    ? input.predecessor_wake_history.filter((row) => row && typeof row === 'object').slice(-MAX_WAKE_HISTORY).map(clone)
   128→    : [];
   129→  return {
   130→    ...base,
   131→    supervisor_epoch: Math.max(1, Number(input.supervisor_epoch) || 1),
   132→    cycle_seq: Math.max(0, Number(input.cycle_seq) || 0),
   133→    state: input.paused === true ? 'PAUSED' : (KEEPALIVE_STATES.includes(input.state) ? input.state : 'RECOVERING'),
   134→    conversation_url: conversationUrl,
   135→    tab_id: input.tab_id ? String(input.tab_id) : null,
   136→    paused: input.paused === true,
   137→    paused_from_state: KEEPALIVE_STATES.includes(input.paused_from_state) ? input.paused_from_state : null,
   138→    process_incarnation_id: sanitizeProcessIncarnationId(input.process_incarnation_id),
   139→    process_incarnation_started_at: input.process_incarnation_started_at || null,
   140→    predecessor_process_incarnation_id: sanitizeProcessIncarnationId(input.predecessor_process_incarnation_id),
   141→    predecessor_fenced_at: input.predecessor_fenced_at || null,
   142→    predecessor_queued_wake_count: Math.max(0, Number(input.predecessor_queued_wake_count) || 0),
   143→    admission_state: ADMISSION_STATES.has(String(input.admission_state || '').toUpperCase())
   144→      ? String(input.admission_state).toUpperCase()
   145→      : 'UNKNOWN',
   146→    admission_reason: input.admission_reason ? String(input.admission_reason).slice(0, 160) : null,
   147→    admission_generation_floor: typeof input.admission_generation_floor === 'number'
   148→      && Number.isSafeInteger(input.admission_generation_floor)
   149→      ? Math.max(0, input.admission_generation_floor)
   150→      : null,
   151→    admission_refill_enabled: typeof input.admission_refill_enabled === 'boolean' ? input.admission_refill_enabled : null,
   152→    admission_supervisor_enabled: typeof input.admission_supervisor_enabled === 'boolean' ? input.admission_supervisor_enabled : null,
   153→    admission_observed_at: input.admission_observed_at || null,
   154→    parked_at: input.parked_at || null,
   155→    parked_reason: input.parked_reason ? String(input.parked_reason).slice(0, 160) : null,
   156→    parked_queued_wake_count: Math.max(0, Number(input.parked_queued_wake_count) || 0),
   157→    parked_wake_reasons: Array.isArray(input.parked_wake_reasons)
   158→      ? input.parked_wake_reasons.map((value) => String(value).slice(0, 80)).slice(-MAX_QUEUED_WAKES)
   159→      : [],
   160→    suppressed_wake_count: Math.max(0, Number(input.suppressed_wake_count) || 0),
   161→    last_suppressed_wake_at: input.last_suppressed_wake_at || null,
   162→    last_suppressed_wake_reason: input.last_suppressed_wake_reason ? String(input.last_suppressed_wake_reason).slice(0, 80) : null,
   163→    queued_wakes: queued,
   164→    pending_wake: input.pending_wake && typeof input.pending_wake === 'object' ? clone(input.pending_wake) : null,
   165→    active_wake: sanitizeActiveWake(input.active_wake),
   166→    predecessor_wake_history: predecessorWakeHistory,
   167→    ambiguous_history: ambiguousHistory,
   168→    last_wake_at: input.last_wake_at || null,
   169→    last_wake_reason: input.last_wake_reason || null,
   170→    last_unsent_attempt_at: input.last_unsent_attempt_at || null,
   171→    last_completed_cycle_at: input.last_completed_cycle_at || null,
   172→    last_research_wake_at: input.last_research_wake_at || null,
   173→    previous_worker_generation: sanitizeWorkerGenerationMap(input.previous_worker_generation),
   174→    rollover_reason: input.rollover_reason ? String(input.rollover_reason).slice(0, 160) : null,
   175→    rollover_release_at: input.rollover_release_at || null,
   176→    rollover_attempt: input.rollover_attempt && typeof input.rollover_attempt === 'object' ? clone(input.rollover_attempt) : null,
   177→    updated_at: input.updated_at || null,
   178→  };
   179→}

   222→export class SupervisorKeepalive {
   223→  #load; #save; #clock; #uuid; #state; #minWakeIntervalMs; #maxCyclesPerEpoch; #processIncarnationId;
   224→
   225→  constructor({ loadState, saveState, clock = () => Date.now(), uuid = () => crypto.randomUUID(), processIncarnationId = null, minWakeIntervalMs = 60000, maxCyclesPerEpoch = null } = {}) {
   226→    if (typeof loadState !== 'function' || typeof saveState !== 'function') throw new Error('keepalive_persistence_required');
   227→    this.#load = loadState;
   228→    this.#save = saveState;
   229→    this.#clock = clock;
   230→    this.#uuid = uuid;
   231→    this.#processIncarnationId = sanitizeProcessIncarnationId(processIncarnationId)
   232→      || `process_${crypto.randomUUID().toLowerCase()}`;
   233→    this.#minWakeIntervalMs = Math.max(30000, Number(minWakeIntervalMs) || 60000);
   234→    void maxCyclesPerEpoch;
   235→    this.#maxCyclesPerEpoch = null;
   236→    this.#state = freshState();
   237→  }
   238→
   239→  async init() {
   240→    this.#state = sanitize(await this.#load());
   241→    const recoveredAt = iso(this.#clock);
   242→    const predecessorIncarnation = this.#state.process_incarnation_id;
   243→    const crossedProcessBoundary = predecessorIncarnation !== this.#processIncarnationId;
   244→
   245→    if (crossedProcessBoundary) {
   246→      const hasPredecessor = predecessorIncarnation != null;
   247→      const pendingProcess = sanitizeProcessIncarnationId(this.#state.pending_wake?.process_incarnation_id);
   248→      if (hasPredecessor
   249→        && this.#state.pending_wake?.ambiguous_at
   250→        && this.#state.pending_wake?.automatic_retry_allowed === false
   251→        && pendingProcess
   252→        && pendingProcess !== this.#processIncarnationId
   253→        && !this.#state.pending_wake.process_boundary_fenced_at) {
   254→        // Persist the safety fact on the ambiguous wake itself. The immediate
   255→        // predecessor pointer is intentionally one-hop and is overwritten on
   256→        // every restart; this wake-local fence survives any number of later
   257→        // process boundaries without ever making the old effect retryable.
   258→        this.#state.pending_wake.process_boundary_fenced_at = recoveredAt;
   259→        this.#state.pending_wake.process_boundary_fenced_by_process_incarnation_id = predecessorIncarnation;
   260→      }
   261→      const predecessorActive = hasPredecessor ? this.#state.active_wake : null;
   262→      if (predecessorActive) {
   263→        this.#state.predecessor_wake_history = [
   264→          ...this.#state.predecessor_wake_history,
   265→          {
   266→            ...clone(predecessorActive),
   267→            retired_at: recoveredAt,
   268→            retired_reason: 'PROCESS_BOUNDARY_ACTIVE_WAKE_FENCED',
   269→            automatic_retry_allowed: false,
   270→            authority_effect: false,
   271→          },
   272→        ].slice(-MAX_WAKE_HISTORY);
   273→      }
   274→      if (hasPredecessor) {
   275→        this.#state.predecessor_process_incarnation_id = predecessorIncarnation;
   276→        this.#state.predecessor_fenced_at = recoveredAt;
   277→        this.#state.predecessor_queued_wake_count = this.#state.queued_wakes.length;
   278→      } else {
   279→        this.#state.predecessor_process_incarnation_id = null;
   280→        this.#state.predecessor_fenced_at = null;
   281→        this.#state.predecessor_queued_wake_count = 0;
   282→      }
   283→      this.#state.queued_wakes = [];
   284→      this.#state.active_wake = null;
   285→      this.#state.previous_worker_generation = {};
   286→      this.#state.process_incarnation_id = this.#processIncarnationId;
   287→      this.#state.process_incarnation_started_at = recoveredAt;
   288→    }
   289→
   290→    if (this.#state.pending_wake && !this.#state.pending_wake.ambiguous_at) {
   291→      this.#state.pending_wake.ambiguous_at = recoveredAt;
   292→      this.#state.pending_wake.ambiguous_reason = 'PROCESS_RESTART_WITH_UNRESOLVED_WAKE_EFFECT';
   293→      this.#state.pending_wake.restart_recovered_at = recoveredAt;
   294→      this.#state.pending_wake.automatic_retry_allowed = false;
   295→    }
   296→    if (this.#state.state === 'ROLLOVER_PENDING') {
   297→      this.#state.state = 'ROLLOVER_AMBIGUOUS';
   298→      this.#state.rollover_reason = 'PROCESS_RESTART_WITH_UNRESOLVED_ROLLOVER_EFFECT';
   299→      this.#state.rollover_attempt = {
   300→        ...(this.#state.rollover_attempt || {}),
   301→        restart_recovered_at: recoveredAt,
   302→        automatic_retry_allowed: false,
   303→      };
   304→    }
   305→    if (crossedProcessBoundary && this.#state.admission_state === 'OPEN') {
   306→      // An OPEN decision is never trusted across a process boundary. A fresh
   307→      // authoritative heartbeat must reopen continuous-service actuation.
   308→      this.#state.admission_state = 'UNKNOWN';
   309→      this.#state.admission_reason = 'PROCESS_RESTART_REOBSERVATION_REQUIRED';
   310→    }
   311→    if (this.#state.paused) this.#state.state = 'PAUSED';
   312→    else if (this.#state.pending_wake) this.#state.state = 'WAKE_AMBIGUOUS';
   313→    else if (['ROLLOVER_DEFERRED','ROLLOVER_REQUIRED','ROLLOVER_AMBIGUOUS'].includes(this.#state.state)) {}
   314→    else if (this.#state.active_wake) this.#state.state = 'ACTIVE';
   315→    else if (this.#state.admission_state === 'CLOSED') this.#state.state = 'PARKED';
   316→    else this.#state.state = this.#state.conversation_url ? 'WAITING' : 'RECOVERING';
   317→    await this.#persist();
   318→    return this.snapshot();
   319→  }
   320→
   321→  snapshot() {
   322→    return Object.freeze({
   323→      ...clone(this.#state),
   324→      work_cycle_limit: this.#maxCyclesPerEpoch,
   325→      worker_generation_memory_limit: MAX_WORKER_GENERATION_MEMORY,
   326→      automatic_rollover_cycle_limit_enabled: false,
   327→      external_confirmation_required_for_continuation: false,
   328→    });
   329→  }
   330→  activeWake() { return clone(this.#state.active_wake); }
   331→
   332→  async applyAdmissionClosed(control = {}) {
   333→    if (control?.authoritative !== true
   334→      || control?.state !== 'CLOSED'
   335→      || control?.continuous_service_allowed !== false
   336→      || typeof control?.refill_enabled !== 'boolean'
   337→      || typeof control?.supervisor_admission_enabled !== 'boolean'
   338→      || typeof control?.generation_floor !== 'number'
   339→      || !Number.isSafeInteger(control.generation_floor)) {
   340→      throw new Error('keepalive_admission_closed_readback_invalid');
   341→    }
   342→    if (this.#state.admission_generation_floor != null
   343→      && control.generation_floor < this.#state.admission_generation_floor) {
   344→      throw new Error('keepalive_admission_generation_floor_regression');
   345→    }
   346→    const observedAt = iso(this.#clock);
   347→    this.#state.admission_state = 'CLOSED';
   348→    this.#state.admission_reason = String(control.reason || 'CONTINUOUS_SERVICE_ADMISSION_FENCED').slice(0, 160);
   349→    this.#state.admission_generation_floor = Math.max(0, control.generation_floor);
   350→    this.#state.admission_refill_enabled = control.refill_enabled;
   351→    this.#state.admission_supervisor_enabled = control.supervisor_admission_enabled;
   352→    this.#state.admission_observed_at = observedAt;
   353→
   354→    if (this.#state.pending_wake) {
   355→      if (!this.#state.pending_wake.ambiguous_at) {
   356→        this.#state.pending_wake.ambiguous_at = observedAt;
   357→        this.#state.pending_wake.ambiguous_reason = 'ADMISSION_CLOSED_WITH_UNRESOLVED_WAKE_EFFECT';
   358→        this.#state.pending_wake.automatic_retry_allowed = false;
   359→      }
   360→      this.#state.state = 'WAKE_AMBIGUOUS';
   361→    } else if (this.#state.active_wake
   362→      || ['ROLLOVER_PENDING','ROLLOVER_AMBIGUOUS'].includes(this.#state.state)) {
   363→      // Preserve already-started or ambiguous effects. The lifecycle can still
   364→      // observe them read-only, but admission-aware canActuate() blocks follow-up.
   365→    } else {
   366→      const queued = this.#state.queued_wakes;
   367→      this.#state.parked_at = observedAt;
   368→      this.#state.parked_reason = this.#state.admission_reason;
   369→      this.#state.parked_queued_wake_count = queued.length;
   370→      this.#state.parked_wake_reasons = [...new Set(queued.map((row) => String(row?.reason || '')).filter(Boolean))];
   371→      this.#state.queued_wakes = [];
   372→      // A released-but-not-started rollover is still only queued intent. Retire
   373→      // it together with wakes so a later pause/resume cannot resurrect it.
   374→      this.#state.rollover_reason = null;
   375→      this.#state.rollover_release_at = null;
   376→      this.#state.rollover_attempt = null;
   377→      this.#state.state = this.#state.paused ? 'PAUSED' : 'PARKED';
   378→    }
   379→    await this.#persist();
   380→    return this.snapshot();
   381→  }
   382→
   383→  async applyAdmissionOpen(control = {}) {
   384→    if (control?.authoritative !== true
   385→      || control?.state !== 'OPEN'
   386→      || control?.continuous_service_allowed !== true
   387→      || control?.refill_enabled !== true
   388→      || control?.supervisor_admission_enabled !== true
   389→      || typeof control?.generation_floor !== 'number'
   390→      || !Number.isSafeInteger(control.generation_floor)) {
   391→      throw new Error('keepalive_admission_open_readback_invalid');
   392→    }
   393→    if (this.#state.admission_generation_floor != null
   394→      && control.generation_floor < this.#state.admission_generation_floor) {
   395→      throw new Error('keepalive_admission_generation_floor_regression');
   396→    }
   397→    this.#state.admission_state = 'OPEN';
   398→    this.#state.admission_reason = null;
   399→    this.#state.admission_generation_floor = Math.max(0, control.generation_floor);

   486→  async enqueueWake(reason, metadata = {}) {
   487→    const normalizedReason = String(reason || '');
   488→    if (!WAKE_REASONS.has(normalizedReason)) throw new Error('keepalive_wake_reason_invalid');
   489→    if (this.#state.admission_state === 'CLOSED') {
   490→      this.#state.suppressed_wake_count += 1;
   491→      this.#state.last_suppressed_wake_at = iso(this.#clock);
   492→      this.#state.last_suppressed_wake_reason = normalizedReason;
   493→      await this.#persist();
   494→      return this.snapshot();
   495→    }
   496→    const key = `${normalizedReason}:${String(metadata.agent_id || metadata.key || '')}`;
   497→    const existing = this.#state.queued_wakes.find((row) => row.key === key && row.process_incarnation_id === this.#processIncarnationId);
   498→    if (existing) {
   499→      const incoming = Array.isArray(metadata.agent_ids) ? metadata.agent_ids.map(String) : [];
   500→      const prior = Array.isArray(existing.metadata?.agent_ids) ? existing.metadata.agent_ids.map(String) : [];
   501→      if (incoming.length > 0) {
   502→        existing.metadata = { ...existing.metadata, ...clone(metadata), agent_ids: [...new Set([...prior, ...incoming])] };
   503→        await this.#persist();
   504→      }
   505→      return this.snapshot();
   506→    }
   507→    this.#state.queued_wakes.push({
   508→      key,
   509→      reason: normalizedReason,
   510→      metadata: clone(metadata),
   511→      queued_at: iso(this.#clock),
   512→      process_incarnation_id: this.#processIncarnationId,
   513→    });
   514→    this.#state.queued_wakes = this.#state.queued_wakes.slice(-MAX_QUEUED_WAKES);
   515→    if (normalizedReason === 'RESEARCH_ACCELERATOR_DUE') this.#state.last_research_wake_at = iso(this.#clock);
   516→    await this.#persist();
   517→    return this.snapshot();
   518→  }

   635→  async observeWorkers(workerSignals = []) {
   636→    const events = [];
   637→    const previous = this.#state.previous_worker_generation;
   638→    const next = {};
   639→    for (const signal of workerSignals) {
   640→      const id = String(signal?.agent_id || '');
   641→      if (!id) continue;
   642→      const lifecycle = String(signal?.lifecycle_state || '');
   643→      const generation = String(signal?.generation_state || 'UNKNOWN').toUpperCase();
   644→      const prev = String(previous[id] || 'UNKNOWN');
   645→      if (prev === 'GENERATING' && generation === 'IDLE') events.push({ reason: 'WORKER_RESULT_READY', agent_id: id });
   646→      if (lifecycle === 'LOST' && prev !== 'TERMINAL') events.push({ reason: 'WORKER_LOST', agent_id: id });
   647→      if (lifecycle === 'PROVISIONING_AMBIGUOUS' && prev !== 'TERMINAL') events.push({ reason: 'WORKER_FAILED', agent_id: id });
   648→      next[id] = generation;
   649→    }
   650→    this.#state.previous_worker_generation = sanitizeWorkerGenerationMap(next);
   651→    await this.#persist();
   652→    for (const event of events.filter((row) => row.reason === 'WORKER_RESULT_READY')) await this.enqueueWake(event.reason, { agent_id: event.agent_id });
   653→    for (const reason of ['WORKER_LOST','WORKER_FAILED']) {
   654→      const agentIds = events.filter((row) => row.reason === reason).map((row) => row.agent_id);
   655→      if (agentIds.length) await this.enqueueWake(reason, { key: 'fleet-terminal-burst', agent_ids: agentIds });
   656→    }
   657→    return events;
   658→  }

   660→  canWake() {
   661→    if (this.#state.admission_state === 'CLOSED'
   662→      || this.#state.paused
   663→      || ['PARKED','WAKE_AMBIGUOUS','ROLLOVER_DEFERRED','ROLLOVER_REQUIRED','ROLLOVER_PENDING','ROLLOVER_AMBIGUOUS'].includes(this.#state.state)) return false;
   664→    const actionable = this.#state.queued_wakes.filter((row) => row.process_incarnation_id === this.#processIncarnationId);
   665→    if (!this.#state.conversation_url || this.#state.pending_wake || this.#state.active_wake || actionable.length === 0) return false;
   666→    // D-K5 (live 2026-09-19): a wake that was prepared but provably NOT sent
   667→    // (pre-effect throw: composer resolution, replace unverified, stale ref)
   668→    // used to be re-prepared on the very next monitor tick — and the
   669→    // CONTINUE_DEVELOPMENT reason bypasses the wake interval entirely, so a
   670→    // permanent pre-effect failure became a ~2s retry storm that raced the
   671→    // page's draft-sync and appended ~3k chars of wake text per MINUTE to the
   672→    // composer. One unsent attempt must cost at least the wake interval.
   673→    if (this.#state.last_unsent_attempt_at) {
   674→      const since = this.#clock() - new Date(this.#state.last_unsent_attempt_at).getTime();
   675→      if (Number.isFinite(since) && since < this.#minWakeIntervalMs) return false;
   676→    }
   677→    if (!this.#state.last_wake_at) return true;
   678→    if (String(actionable[0]?.reason || '') === 'CONTINUE_DEVELOPMENT') return true;
   679→    return this.#clock() - new Date(this.#state.last_wake_at).getTime() >= this.#minWakeIntervalMs;
   680→  }

   781→  async retireAmbiguousAfterProcessBoundary({ reason = 'PROCESS_BOUNDARY_ORIGINAL_BOOTSTRAP_TARGET_LOST', replacement_tab_id = null } = {}) {
   782→    const pending = this.#state.pending_wake;
   783→    if (!pending || !pending.ambiguous_at) throw new Error('keepalive_no_ambiguous_wake');
   784→    const pendingProcess = sanitizeProcessIncarnationId(pending.process_incarnation_id);
   785→    const currentProcess = sanitizeProcessIncarnationId(this.#state.process_incarnation_id);
   786→    const predecessorProcess = sanitizeProcessIncarnationId(this.#state.predecessor_process_incarnation_id);
   787→    if (!pendingProcess || !currentProcess || pendingProcess === currentProcess) {
   788→      throw new Error('keepalive_process_boundary_ambiguity_not_proven');
   789→    }
   790→    const immediatePredecessorFence = predecessorProcess === pendingProcess
   791→      && Boolean(this.#state.predecessor_fenced_at);
   792→    const durableWakeBoundaryFence = Boolean(pending.process_boundary_fenced_at)
   793→      && pending.automatic_retry_allowed === false;
   794→    if (!this.#state.predecessor_fenced_at || (!immediatePredecessorFence && !durableWakeBoundaryFence)) {
   795→      throw new Error('keepalive_process_boundary_predecessor_not_proven');
   796→    }
   797→    const retiredAt = iso(this.#clock);
   798→    this.#state.cycle_seq = Math.max(this.#state.cycle_seq, Math.max(1, Number(pending.cycle_seq) || 1));
   799→    this.#state.queued_wakes = this.#state.queued_wakes.filter((row) => !(
   800→      row.key === pending.queue_key
   801→      && sanitizeProcessIncarnationId(row.process_incarnation_id) === pendingProcess
   802→    ));
   803→    this.#state.ambiguous_history = [
   804→      ...this.#state.ambiguous_history,
   805→      {
   806→        ...clone(pending),
   807→        retired_at: retiredAt,
   808→        retired_reason: String(reason || 'PROCESS_BOUNDARY_ORIGINAL_BOOTSTRAP_TARGET_LOST').slice(0, 200),
   809→        retired_process_incarnation_id: pendingProcess,
   810→        recovery_process_incarnation_id: currentProcess,
   811→        replacement_tab_id: replacement_tab_id ? String(replacement_tab_id).slice(0, 120) : null,
   812→        automatic_retry_allowed: false,
   813→      },
   814→    ].slice(-MAX_WAKE_HISTORY);
   815→    this.#state.pending_wake = null;
   816→    this.#state.last_completed_cycle_at = retiredAt;
   817→    this.#state.state = this.#postWakeSettlementState('RECOVERING');
   818→    await this.#persist();
   819→    return this.snapshot();
   820→  }

   822→  async retireAmbiguousAfterTerminal({ tab_id = null, generation_epoch = null, reason = 'TERMINAL_BOUNDARY_CONFIRMED' } = {}) {
   823→    const pending = this.#state.pending_wake;
   824→    if (!pending || !pending.ambiguous_at) throw new Error('keepalive_no_ambiguous_wake');
   825→    if (tab_id != null && this.#state.tab_id != null && String(tab_id) !== String(this.#state.tab_id)) throw new Error('keepalive_terminal_tab_binding_mismatch');
   826→    const epoch = generation_epoch == null ? null : Number(generation_epoch);
   827→    if (epoch != null && (!Number.isSafeInteger(epoch) || epoch < 0)) throw new Error('keepalive_terminal_generation_invalid');
   828→    const retiredAt = iso(this.#clock);
   829→    this.#state.cycle_seq = Math.max(this.#state.cycle_seq, Math.max(1, Number(pending.cycle_seq) || 1));
   830→    this.#state.queued_wakes = this.#state.queued_wakes.filter((row) => row.key !== pending.queue_key);
   831→    this.#state.ambiguous_history = [
   832→      ...this.#state.ambiguous_history,
   833→      {
   834→        ...clone(pending),
   835→        retired_at: retiredAt,
   836→        retired_reason: String(reason || 'TERMINAL_BOUNDARY_CONFIRMED').slice(0, 200),
   837→        terminal_generation_epoch: epoch,
   838→        automatic_retry_allowed: false,
   839→      },
   840→    ].slice(-MAX_WAKE_HISTORY);
   841→    this.#state.pending_wake = null;
   842→    this.#state.last_completed_cycle_at = retiredAt;
   843→    this.#state.state = this.#postWakeSettlementState();
   844→    await this.#persist();
   845→    return this.snapshot();
   846→  }

   848→  async resolveAmbiguous({ observed_sent } = {}) {
   849→    if (this.#state.state !== 'WAKE_AMBIGUOUS' || !this.#state.pending_wake) throw new Error('keepalive_no_ambiguous_wake');
   850→    if (observed_sent === true) return this.confirmWakeSent(this.#state.pending_wake.wake_id);
   851→    if (observed_sent === false) {
   852→      // D-K5: the wake was provably not sent — record the attempt so the
   853→      // retry pays the wake-interval price instead of storming.
   854→      this.#state.last_unsent_attempt_at = iso(this.#clock);
   855→      this.#state.pending_wake = null;
   856→      this.#state.state = this.#postWakeSettlementState();
   857→      await this.#persist();
   858→      return this.snapshot();
   859→    }
   860→    throw new Error('keepalive_ambiguous_resolution_requires_observation');
   861→  }
   862→
   863→  // LIVE 2026-09-21 (self-update restart window): a wake that went
   864→  // TYPE_EFFECT_AMBIGUOUS exactly across the successor boot can fence the
   865→  // rollover FOREVER — the lifecycle's rollover retire paths only accept a
   866→  // pending wake in WAKE_AMBIGUOUS, so ROLLOVER_REQUIRED + pending-ambiguous
   867→  // deadlocks the lifecycle (keepalive_no_ambiguous_wake on every tick) and
   868→  // the idle maintenance window never idles, which starves the devos task
   869→  // cycle (no leases). This settlement is proof-based, from ROLLOVER_REQUIRED:
   870→  //   observed_sent === true  → the marker is provably in the transcript:
   871→  //                             confirm the wake (resume normal processing).
   872→  //   observed_sent === false → the wake provably did not cross (composer
   873→  //                             still holds the exact message / terminal and
   874→  //                             no marker): drop it; postWakeSettlementState
   875→  //                             PRESERVES the ROLLOVER_* state so the
   876→  //                             rollover proceeds on a clean ledger, and the
   877→  //                             wake re-prepares later at the wake-interval
   878→  //                             price (D-K5 anti-storm).
   879→  async settleRolloverBlockedAmbiguousWake({ observed_sent } = {}) {
   880→    const pending = this.#state.pending_wake;
   881→    if (this.#state.state !== 'ROLLOVER_REQUIRED' || !pending?.ambiguous_at) throw new Error('keepalive_no_rollover_blocked_ambiguous_wake');
   882→    if (observed_sent === true) return this.confirmWakeSent(pending.wake_id);
   883→    if (observed_sent === false) {
   884→      this.#state.last_unsent_attempt_at = iso(this.#clock);
   885→      this.#state.pending_wake = null;
   886→      this.#state.state = this.#postWakeSettlementState();
   887→      await this.#persist();
   888→      return this.snapshot();
   889→    }
   890→    throw new Error('keepalive_ambiguous_resolution_requires_observation');
   891→  }

   893→  async #persist() {
   894→    this.#state.version = SUPERVISOR_KEEPALIVE_VERSION;
   895→    this.#state.updated_at = iso(this.#clock);
   896→    await this.#save(clone(this.#state));
   897→  }
