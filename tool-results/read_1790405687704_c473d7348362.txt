    24→async function jfetch<T>(path: string, init?: RequestInit): Promise<T> {
    25→  const res = await fetch(dpath(path), { cache: 'no-store', ...init })
    26→  const data: unknown = await res.json().catch(() => null)
    27→  if (!res.ok) {
    28→    const err = data as { error?: { message?: string; code?: string } } | null
    29→    throw new Error(err?.error?.message ? `${err.error.message} [${err.error.code}]` : `HTTP ${res.status}`)
    30→  }
    31→  return data as T
    32→}

    34→function humanS(s: number | null | undefined): string {
    35→  if (s == null || s < 0) return '—'
    36→  if (s < 60) return `${s}s`
    37→  if (s < 3600) return `${Math.floor(s / 60)}m ${s % 60}s`
    38→  if (s < 86400) return `${(s / 3600).toFixed(1)}h`
    39→  return `${(s / 86400).toFixed(1)}d`
    40→}

    42→function hhmmss(iso: string): string {
    43→  const d = new Date(iso)
