   180→export function planPostRestoreBlankTabCleanup({ continuityRow, bindings = [], currentTabs = [] } = {}) {
   181→  const desiredRootCount = (continuityRow?.tabs || []).filter((tab) => isAgentPlatformRoot(tab?.url)).length;
   182→  const boundTabIds = new Set((bindings || []).map((row) => String(row?.tab_id || '')).filter(Boolean));
   183→  let retainedRoots = (currentTabs || []).filter((tab) => boundTabIds.has(String(tab?.tab_id || '')) && isAgentPlatformRoot(tab?.url)).length;
   184→  const candidates = (currentTabs || [])
   185→    .filter((tab) => !boundTabIds.has(String(tab?.tab_id || '')) && isAgentPlatformRoot(tab?.url))
   186→    .sort((a, b) => Number(a?.selected === true) - Number(b?.selected === true));
   187→  const closeTabIds = [];
   188→  for (const tab of candidates) {
   189→    if (retainedRoots < desiredRootCount) {
   190→      retainedRoots += 1;
   191→      continue;
   192→    }
   193→    const tabId = String(tab?.tab_id || '');
   194→    if (tabId) closeTabIds.push(tabId);
   195→  }
   196→  return Object.freeze({
   197→    close_tab_ids: closeTabIds,
   198→    desired_root_count: desiredRootCount,
   199→    bound_tab_count: boundTabIds.size,
   200→    current_root_count: (currentTabs || []).filter((tab) => isAgentPlatformRoot(tab?.url)).length,
   201→    arbitrary_tab_close: false,
   202→    authority_effect: false,
   203→  });
   204→}

   222→// D-L2: pure classification of legitimate no-op completions. Exported for
   223→// contract tests; returns null when the action still requires the generic
   224→// postcondition readback path.
   225→export function classifyNoOpEffectOutcome(actionRaw, result) {
   226→  const action = String(actionRaw || '').toUpperCase();
   227→  if (action === 'BACK' || action === 'FORWARD') {
   228→    if (result && result.navigated === false) return 'NO_EFFECT_PROVEN';
   229→    if (result && result.navigated === true) return 'CONFIRMED';
   230→    return null;
   231→  }
   232→  if (action === 'RELOAD') {
   233→    // webContents.reload() deterministically initiates the reload; a failed
   234→    // load surfaces through perception_error afterwards.
   235→    return 'CONFIRMED';
   236→  }
   237→  if (action === 'SCROLL') {
   238→    const scroll = result?.scroll;
   239→    if (scroll && scroll.proof === 'VIEWPORT_PAGE_Y_CHANGED') return 'CONFIRMED';
   240→    if (scroll && scroll.proof === 'SCROLL_BOUNDARY_REACHED') return 'NO_EFFECT_PROVEN';
   241→    return null;
   242→  }
   243→  if (action === 'SELF_UPDATE_CHECK') {
   244→    const state = String(result?.state || '').toUpperCase();
   245→    if (state === 'CURRENT') return 'NO_EFFECT_PROVEN';
   246→    return 'CONFIRMED';
   247→  }
   248→  return null;
   249→}

   441→    };
   442→
   443→    this.#lifecycle = new SupervisorLifecycleRuntime({

   454→      ...(meshStatePath ? { statePath: meshStatePath } : {}),
   455→    });

   461→      // to be terminated when the 180-second startup grace expires.
   462→      hostResilience,
   463→      canRestart: async () => {
   464→        if (!controlModeAllows(this.#supervisorMode)) return false;
   465→        if (!armedAllows(this.#armed)) return false;
   466→        if (this.#currentCommands.size > 0 && !globalOwnerGateDisabled('self_update.current_command')) return false;
   467→        return confirmSelfUpdateRestartSafety({ getState: this.#getState });
   468→      },
   469→      beforeInstall: async (receipt) => {
   470→        const { app } = await import('electron');
   471→        await persistPreInstallReceipt(app, receipt);
   472→      },
   473→      readPriorTransaction: async () => {
   474→        const { app } = await import('electron');
   475→        return readSelfUpdateTransaction(app).catch(() => null);
   476→      },
   477→      beforeInstallerLaunch: async (receipt) => {

   479→        if (!app?.isPackaged && !globalOwnerGateDisabled('self_update.packaged_required')) throw new Error('native_supervisor_self_update_packaged_required');
   480→        if (!app.hasSingleInstanceLock() && !globalOwnerGateDisabled('self_update.primary_instance_lock')) throw new Error('native_supervisor_self_update_primary_lock_required');
   481→        await this.#persistSessionContinuity(app, receipt);
   482→        await beforeSelfUpdateInstall?.(structuredClone(receipt));

   636→  async #restoreSessionContinuity() {
   637→    const { app } = await import('electron');
   638→    const userData = app.getPath('userData');
   639→    // P0 repair (point 2) — ONE-SHOT FENCE: the capsule is claimed by a durable
   640→    // rename BEFORE any NEW_TAB is issued. A process that crashes mid-restore,
   641→    // and every successor process, will find the canonical path absent: the
   642→    // 7 -> 14 -> 21 -> 28 -> 32 replay amplifier of the 2026-09-17 incident is
   643→    // structurally impossible with this fence in place.
   644→    const attempt = await beginSelfUpdateSessionContinuityRestoreAttempt(userData);
   645→    if (!attempt) {
   646→      this.#continuityStatus = { state: 'NONE', restored_tabs: 0, target_version: null, user_session_continuity: null, tab_cardinality_continuity: null, authority_effect: false };
   647→      return null;
   648→    }
   649→    const { row } = attempt;
   650→    this.#continuityStatus = { state: 'FOUND', restored_tabs: 0, tab_count: row.tabs.length, target_version: row.target_version || null, user_session_continuity: null, tab_cardinality_continuity: null, authority_effect: false };
   651→    if (row.target_version && String(row.target_version) !== this.#version) {
   652→      this.#continuityStatus.state = 'TARGET_VERSION_MISMATCH';
   653→      return row;
   654→    }
   655→
   656→    // P0 repair (points 2+3): the module-level restore is auth-aware and
   657→    // one-shot. The first restored /c/ tab observed on /auth/login latches
   658→    // AUTH_REQUIRED as a TERMINAL state — remaining ChatGPT tabs are skipped
   659→    // (no login-page multiplication), non-ChatGPT tabs still restore, and the
   660→    // capsule is already fenced so no process will ever replay it.
   661→    const restore = await restoreSelfUpdateSessionContinuity({
   662→      row,
   663→      currentVersion: this.#version,
   664→      getState: this.#getState,
   665→      executeCommand: this.#executeCommand,
   666→    });
   667→
   668→    // P0 repair (point 4) — cleanup by proof only: close tabs whose
   669→    // created_by_continuity_id matches THIS attempt beyond the capsule's own
   670→    // tab count. Tabs without the stamp (including the 32 legacy live tabs
   671→    // of the incident) are never touched.
   672→    let closedExtraTabs = 0;
   673→    try {
   674→      const postRestoreState = await this.#getState();
   675→      const cleanup = planPostRestoreDuplicateTabCleanup({ continuityRow: row, currentTabs: postRestoreState?.tabs || [] });
   676→      for (const tabId of cleanup.close_tab_ids) {
   677→        await this.#executeCommand({ action: 'CLOSE_TAB', payload: { tab_id: tabId }, platform: null });
   678→        closedExtraTabs += 1;
   679→      }
   680→    } catch {
   681→      // Best-effort: un-closed duplicates remain observable via census.
   682→    }
   683→
   684→    let reconcile = { schema: 'metaengine.self-update-chat-reconcile.v1', tabs: [], ambiguous_count: 0, unresolved_count: 0, authority_effect: false };
   685→    if (restore.state === 'RESTORED' && restore.bindings.some((binding) => binding.generation_state === 'GENERATING')) {
   686→      reconcile = await reconcileRestoredGeneratingChats({
   687→        bindings: restore.bindings,
   688→        captureTab: async (tabId) => this.#executeCommand({ action: 'CAPTURE', payload: { tab_id: String(tabId) }, platform: AGENT_PLATFORM_ID }),
   689→        clickControl: async (tabId, control) => this.#executeCommand({
   690→          action: 'TYPED_CLICK',
   691→          payload: {
   692→            tab_id: String(tabId),
   693→            role: 'button',
   694→            accessible_name: String(control?.name || ''),
   695→            semantic_ref: control?.semantic_ref,
   696→          },
   697→          platform: AGENT_PLATFORM_ID,
   698→        }),
   699→      });
   700→    }
   701→
   702→    const failedTabs = restore.failed_tabs + Number(reconcile.unresolved_count || 0);
   703→    let stateLabel = restore.state;
   704→    if (stateLabel === 'RESTORED' && failedTabs > 0) stateLabel = 'PARTIAL';
   705→    this.#continuityStatus = {
   706→      state: stateLabel, restored_tabs: restore.restored_tabs, closed_extra_tabs: closedExtraTabs, failed_tabs: failedTabs,
   707→      skipped_auth_required_tabs: restore.skipped_auth_required_tabs,
   708→      tab_count: restore.tab_count, target_version: restore.target_version || null, had_generating_tabs: restore.had_generating_tabs,
   709→      lifecycle_resume_present: restore.lifecycle_resume_present, reconciled_generating_tabs: reconcile.tabs.length,
   710→      reconcile_ambiguous_count: reconcile.ambiguous_count, reconcile_unresolved_count: reconcile.unresolved_count,
   711→      reconcile_authority_effect: reconcile.authority_effect === true,
   712→      // Qualification V2 evidence (metadata-only, P0 repair point 5).
   713→      continuity_id: restore.continuity_id,
   714→      auth_readback_state: restore.auth_readback?.auth_state ?? null,
   715→      auth_redirect_tab_count: restore.auth_readback?.auth_redirect_tab_count ?? null,
   716→      user_session_continuity: restore.user_session_continuity,

   719→      tab_cardinality_post_tab_count: restore.tab_cardinality?.post_tab_count ?? null,
   720→      authority_effect: false,

  1061→  // Bounded, idempotent redelivery for command-result transport. The browser
  1062→  // effect has already happened when a receipt is posted; dropping the receipt
  1063→  // on a single transient transport failure turned completed commands into
  1064→  // EXPIRED lease_timeout_no_retry (observed live). Re-POSTing the immutable
  1065→  // receipt is safe — the completion RPC is atomic and idempotent per
  1066→  // command_id — and it never re-executes any browser effect. 4xx responses are
  1067→  // contract violations and are surfaced immediately without retry.
  1068→  static #retryableResultDeliveryFailure(status) {
  1069→    return Number(status) >= 500 || !Number.isFinite(Number(status));
  1070→  }

  1072→  async #deliverResultWithRetry(path, payload) {
  1073→    let lastError = null;
  1074→    for (let attempt = 1; attempt <= this.#resultDeliveryAttempts; attempt += 1) {
  1075→      let response;
  1076→      try {
  1077→        response = await this.#signedRequest(path, { payload });
  1078→      } catch (error) {
  1079→        lastError = error;
  1080→        response = null;
  1081→      }
  1082→      if (response && response.ok) return response;
  1083→      if (response && !NativeSupervisorClient.#retryableResultDeliveryFailure(response.status)) {
  1084→        throw new Error(`native_supervisor_result_http_${response.status}`);
  1085→      }
  1086→      if (response) lastError = new Error(`native_supervisor_result_http_${response.status}`);
  1087→      if (attempt < this.#resultDeliveryAttempts) {
  1088→        const backoff = this.#resultDeliveryBackoffMs[Math.min(attempt - 1, this.#resultDeliveryBackoffMs.length - 1)] || 0;
  1089→        if (backoff > 0) await new Promise((resolve) => setTimeout(resolve, backoff));
  1090→      }
  1091→    }
  1092→    throw lastError || new Error('native_supervisor_result_delivery_failed');
  1093→  }

  1231→  async #executeLocalOrRemote(command) {

  1234→    // enrollment + device-signed lease transport before reaching this function.

  1236→    // immutable release authority and the Guardian durable effect journal.

  1251→        });

  1253→      return this.#developerEmergencyUpdate(structuredClone(command));
  1254→    }
  1255→    if (ROOT_POLICY_ACTIONS.has(action)) return this.#executeCommand(command);
  1256→    if (action === 'ARM') {
  1257→      this.#supervisorMode = 'CONTROL';

  1262→    if (action === 'DISARM') {
  1263→      const error = new Error(ALWAYS_ON_CONTROL_ERROR);
  1264→      error.code = ALWAYS_ON_CONTROL_ERROR;
  1265→      throw error;
  1266→    }
  1267→    if (action === 'SET_SUPERVISOR_MODE') {
  1268→      const next = String(command?.payload?.mode || '').trim().toUpperCase();

  1270→        const error = new Error(ALWAYS_ON_CONTROL_ERROR);
  1271→        error.code = ALWAYS_ON_CONTROL_ERROR;

  1273→      }
  1274→      this.#supervisorMode = 'CONTROL';
  1275→      this.#armed = true;
  1276→      await this.#persistControlState();
  1277→      return { supervisor_mode: 'CONTROL', armed: true, authority_effect: true };
  1278→    }
  1279→    if (action === 'SET_MODE') {
  1280→      const requestedMode = String(command?.payload?.mode || command?.payload?.operator_mode || '').trim().toUpperCase();
  1281→      // SET_MODE is a legacy wire compatibility surface. OBSERVE and GATE_SEND no
  1282→      // longer represent runtime authority states; they canonicalize to the sole
  1283→      // supported always-on state instead of weakening setControlState().
  1284→      if (!['CONTROL','OBSERVE','GATE_SEND'].includes(requestedMode)) {
  1285→        const error = new Error(ALWAYS_ON_CONTROL_ERROR);
  1286→        error.code = ALWAYS_ON_CONTROL_ERROR;
  1287→        throw error;
  1288→      }
  1289→      this.#supervisorMode = 'CONTROL';
  1290→      this.#armed = true;
  1291→      await this.#persistControlState();
  1292→      return { operator_mode: 'CONTROL', supervisor_mode: 'CONTROL', armed: true, authority_effect: true };
  1293→    }
  1294→    if (action === 'CONTROL_CAPABILITIES') return browserControlCapabilities();
  1295→    if (action === 'SELF_UPDATE_STATUS') return this.#selfUpdate?.snapshot() || null;
  1296→    if (action === 'SELF_UPDATE_CHECK') {
  1297→      if (!controlModeAllows(this.#supervisorMode)) throw new Error(`native_supervisor_control_required:${this.#supervisorMode}`);
  1298→      if (!armedAllows(this.#armed)) throw new Error('native_supervisor_disarmed');
  1299→      return this.#selfUpdate?.checkNow();
  1300→    }
  1301→    if (action === 'SELF_UPDATE_APPLY') {
  1302→      if (!controlModeAllows(this.#supervisorMode)) throw new Error(`native_supervisor_control_required:${this.#supervisorMode}`);
  1303→      if (!armedAllows(this.#armed)) throw new Error('native_supervisor_disarmed');
  1304→      return this.#selfUpdate?.applyWhenSafe();
  1305→    }
  1306→    if (!controlModeAllows(this.#supervisorMode) && !READ_ONLY_ACTIONS.has(action)) throw new Error(`native_supervisor_control_required:${this.#supervisorMode}`);
  1307→    if (!armedAllows(this.#armed) && !READ_ONLY_ACTIONS.has(action)) throw new Error('native_supervisor_disarmed');
  1308→    return this.#executeCommand(command);
  1309→  }

  1314→    this.#commandStartsAtMs.set(String(command.command_id), Date.now());
  1315→    if (!this.#currentCommand) this.#currentCommand = projection;
  1316→  }

  1318→  #trackCommandEnd(command) {
  1319→    this.#currentCommands.delete(String(command.command_id));
  1320→    this.#commandStartsAtMs.delete(String(command.command_id));
  1321→    this.#currentCommand = this.#currentCommands.values().next().value || null;
  1322→  }

  1324→  async #effectOutcome(command, result, descriptor, { beforeState = null } = {}) {
  1325→    if (descriptor.read_only) return null;
  1326→    const action = String(command?.action || '').toUpperCase();
  1327→    if (action === 'DOWNLOAD_FILE') {
  1328→      return verifiedDownloadReceiptConfirmsRequest(command?.payload, result) ? 'CONFIRMED' : 'AMBIGUOUS';
  1329→    }
  1330→    const explicit = String(result?.effect_outcome || '').toUpperCase();
  1331→    if (TERMINAL_EFFECT_OUTCOMES.has(explicit)) return explicit;
  1332→    const state = String(result?.effect_state || '').toUpperCase();
  1333→    if (PROVEN_EFFECT_STATES.has(state)) return 'CONFIRMED';
  1334→    if (state.startsWith('AMBIGUOUS')) return 'AMBIGUOUS';
  1335→
  1336→    if (['ARM','SET_SUPERVISOR_MODE','SET_MODE'].includes(action)) return 'CONFIRMED';
  1337→    if (action === 'NEW_TAB' && result?.tab_id) return 'CONFIRMED';
  1338→    if (['FLEET_SET_PROFILE','GATE_DISABLE','GATE_DISABLE_ALL','GATE_ENABLE','GATE_ENABLE_ALL'].includes(action) && result) return 'CONFIRMED';
  1339→    // D-U1 (2026-09-19, live recon): TYPED_CLICK and PRESS_KEY receipts carry
  1340→    // the effect-runtime-bound target they dispatched to (bounded click point
  1341→    // / whitelisted key on an exact semantic target). That dispatch proof is
  1342→    // the strongest available evidence for these operator-recon lanes —
  1343→    // without a classifier they ALWAYS landed in the AMBIGUOUS quarantine
  1344→    // (observed live: postcondition_not_confirmed:AMBIGUOUS on every recon
  1345→    // click/keypress even when the effect ran), which made operator-driven UI
  1346→    // exploration needlessly opaque. Confirmed = the bounded input reached the
  1347→    // bound target; NO_EFFECT_PROVEN stays reserved for explicit no-ops.
  1348→    if ((action === 'TYPED_CLICK' || action === 'PRESS_KEY')
  1349→      && result?.target && (result?.point || result?.key)) return 'CONFIRMED';
  1350→
  1351→    if (action === 'FLEET_RECONCILE') {
  1352→      const before = fleetSnapshotFrom(beforeState);
  1353→      const returned = fleetSnapshotFrom(result);
  1354→      const afterState = await this.#getState().catch(() => null);
  1355→      const after = fleetSnapshotFrom(afterState);
  1356→      const ambiguousAgent = Array.isArray(result?.agents)
  1357→        && result.agents.some((agent) => String(agent?.lifecycle_state || '') === 'PROVISIONING_AMBIGUOUS');
  1358→      if (!before || !returned || !after || ambiguousAgent
  1359→        || !sameValue(projectFleetReconcileSemantics(returned), projectFleetReconcileSemantics(after))) return 'AMBIGUOUS';
  1360→      const explicitPhysicalChange = (Array.isArray(result?.elastic_retired) && result.elastic_retired.length > 0)
  1361→        || (Array.isArray(result?.orphan_fleet_tabs_swept) && result.orphan_fleet_tabs_swept.length > 0);
  1362→      if (explicitPhysicalChange) return 'CONFIRMED';
  1363→      return classifyFleetReconcileOutcome({
  1364→        before,
  1365→        after,
  1366→        active: command?.payload?.active === true,
  1367→        target_agents: command?.payload?.target_agents ?? null,
  1368→      }).effect_outcome;
  1369→    }
  1370→
  1371→    if (['CLOSE_TAB','SELECT_TAB','NAVIGATE'].includes(action)) {
  1372→      const stateReadback = await this.#getState().catch(() => null);
  1373→      const tabId = String(command?.payload?.tab_id || result?.tab_id || '');
  1374→      const tabs = Array.isArray(stateReadback?.tabs) ? stateReadback.tabs : [];
  1375→      if (action === 'CLOSE_TAB') return tabId && !tabs.some((tab) => String(tab?.tab_id || '') === tabId) ? 'CONFIRMED' : 'AMBIGUOUS';
  1376→      if (action === 'SELECT_TAB') return tabId && String(stateReadback?.active_tab?.tab_id || '') === tabId ? 'CONFIRMED' : 'AMBIGUOUS';
  1377→      if (action === 'NAVIGATE') {
  1378→        const row = tabs.find((tab) => String(tab?.tab_id || '') === tabId);
  1379→        return row && result?.url && String(row.url || '') === String(result.url) ? 'CONFIRMED' : 'AMBIGUOUS';
  1380→      }
  1381→    }
  1382→
  1383→    // D-L2: model legitimate no-op completions explicitly instead of quarantining
  1384→    // them as AMBIGUOUS (observed live as FAILED postcondition_not_confirmed on
  1385→    // healthy commands: BACK/FORWARD at history boundary, SCROLL at scroll
  1386→    // boundary, SELF_UPDATE_CHECK while CURRENT, RELOAD initiation).
  1387→    const noOpOutcome = classifyNoOpEffectOutcome(action, result);
  1388→    if (noOpOutcome) return noOpOutcome;
  1389→
  1390→    // Dispatch success is not post-condition proof. Unsupported effect types are
  1391→    // quarantined instead of being mislabeled COMPLETED and are never auto-retried.
  1392→    return 'AMBIGUOUS';
  1393→  }

  1395→  async #executeForLane(command, descriptor) {
  1396→    if (!descriptor.read_only && this.#maintenancePromise) await this.#maintenancePromise;
  1397→    this.#trackCommandStart(command);
  1398→    const started = Date.now();
  1399→    try {
  1400→      const beforeState = String(command?.action || '').toUpperCase() === 'FLEET_RECONCILE'
  1401→        ? await this.#getState().catch(() => null)
  1402→        : null;
  1403→      const result = await this.#executeLocalOrRemote(command);
  1404→      const effectOutcome = await this.#effectOutcome(command, result, descriptor, { beforeState });
  1405→      return { result, effect_outcome: effectOutcome, execution_ms: Date.now() - started };
  1406→    } finally {
  1407→      this.#trackCommandEnd(command);
  1408→    }
  1409→  }

  1411→  async #runCommand(command) {
  1412→    const descriptor = classifyNativeSupervisorCommand(command);
  1413→    let result = null;
  1414→    let effectOutcome = descriptor.read_only ? null : 'AMBIGUOUS';
  1415→    try {
  1416→      const execution = await this.#executeForLane(command, descriptor);
  1417→      result = execution.result;
  1418→      effectOutcome = execution.effect_outcome;
  1419→      await this.#postResult(command, true, result, null, effectOutcome);
  1420→      this.#lastCommandId = command.command_id;
  1421→      this.#lastCommandStatus = descriptor.read_only || ['CONFIRMED','NO_EFFECT_PROVEN'].includes(effectOutcome) ? 'COMPLETED' : 'AMBIGUOUS';
  1422→      return result;
  1423→    } catch (error) {
  1424→      if (String(error?.code || '').startsWith('NATIVE_RESULT_DELIVERY_')) {
  1425→        // The command/effect already ran and the immutable receipt is unresolved.
  1426→        // Never manufacture a second failure receipt and never re-execute the effect.
  1427→        this.#lastCommandId = command.command_id;
  1428→        this.#lastCommandStatus = 'AMBIGUOUS';
  1429→        throw error;
  1430→      }
  1431→      const message = clipError(error);
  1432→      effectOutcome = failedCommandEffectOutcome(command, descriptor, error);
  1433→      await this.#postResult(command, false, result, message, effectOutcome).catch(() => {});
  1434→      this.#lastCommandId = command.command_id;
  1435→      this.#lastCommandStatus = 'FAILED';
  1436→      throw error;
  1437→    }
  1438→  }

  1440→  async #runCommandBatch(commands) {
  1441→    const execution = await this.#commandLane.drain(commands, async (command, descriptor) => {
  1442→      const out = await this.#executeForLane(command, descriptor);
  1443→      return { command, descriptor, ...out };
  1444→    });
  1445→    const rows = execution.map((row, index) => {
  1446→      const nested = row.result || {};
  1447→      const descriptor = classifyNativeSupervisorCommand(commands[index]);
  1448→      return {
  1449→        command: commands[index],
  1450→        descriptor,
  1451→        ok: row.ok,
  1452→        result: row.ok ? nested.result ?? null : null,
  1453→        effect_outcome: row.ok
  1454→          ? nested.effect_outcome ?? null
  1455→          : failedCommandEffectOutcome(commands[index], descriptor, row.error, { schedulerRejected: row.scheduler_rejected === true }),
  1456→        execution_ms: row.ok ? nested.execution_ms ?? row.execution_ms : row.execution_ms,
  1457→        error: row.error,
  1458→      };
  1459→    });
  1460→    await this.#postBatchResults(rows);
  1461→    const last = rows.at(-1) || null;
  1462→    if (last) {
  1463→      this.#lastCommandId = last.command.command_id;
  1464→      this.#lastCommandStatus = last.ok
  1465→        ? (last.descriptor.read_only || ['CONFIRMED','NO_EFFECT_PROVEN'].includes(last.effect_outcome) ? 'COMPLETED' : 'AMBIGUOUS')
  1466→        : 'FAILED';
  1467→    }
  1468→    return rows;
  1469→  }

  1471→  async cycle() {
  1472→    if (this.#legacyFastlaneBusy) return this.snapshot();
  1473→    if (this.#cyclePromise) {
  1474→      // F-L1c: a healthy cycle is bounded by the CDP/result deadlines; a cycle
  1475→      // older than the stall guard means an execution path the bounds did not
  1476→      // reach. Surface it instead of waiting silently (the heartbeat stays alive
  1477→      // in parallel, which previously masked the wedge completely).
  1478→      const cycleAgeMs = this.#cycleStartedAtMs > 0 ? Date.now() - this.#cycleStartedAtMs : 0;
  1479→      if (cycleAgeMs > COMMAND_CYCLE_STALL_GUARD_MS) {
  1480→        this.#lastError = `command_cycle_stall:${cycleAgeMs}ms`;
  1481→      }
  1482→      return this.#cyclePromise;
  1483→    }
  1484→    this.#cycleStartedAtMs = Date.now();
  1485→    this.#cyclePromise = (async () => {
  1486→      try {
  1487→        const identity = await this.ensureEnrollment();
  1488→        if (!identity?.device_id) return this.snapshot();
  1489→
  1490→        // Heartbeat is kept alive independently, but it never sits in front of the
  1491→        // command lease. Its state collection may be expensive (perception, fleet,
  1492→        // mesh), so the fast lane is allowed to proceed while the heartbeat is sent.
  1493→        this.#kickHeartbeat();
  1494→
  1495→        const commands = await this.#nextCommands();
  1496→        if (commands.length > 0) {
  1497→          if (this.#batchTransport === 'SUPPORTED') await this.#runCommandBatch(commands);
  1498→          else await this.#runCommand(commands[0]);
  1499→        } else {
  1500→          // Heavy reconciler work only starts in an idle window. A later mutation
  1501→          // waits for an already-running maintenance pass instead of racing it.
  1502→          this.#kickMaintenance();
  1503→        }
  1504→        this.#lastError = null;
  1505→        return this.snapshot();
  1506→      } catch (error) {
  1507→        this.#lastError = clipError(error);
  1508→        throw error;
  1509→      }
  1510→    })().finally(() => { this.#cyclePromise = null; });
  1511→    return this.#cyclePromise;
  1512→  }
