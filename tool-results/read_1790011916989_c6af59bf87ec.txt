    46→function exactIncarnation(before, after, tabId) {
    47→  const expectedTabId = String(tabId || '');
    48→  const beforeProcess = String(before?.process_incarnation_id || '');
    49→  const afterProcess = String(after?.process_incarnation_id || '');
    50→  const beforeTarget = String(before?.target_id || '');
    51→  const afterTarget = String(after?.target_id || '');
    52→  if (!expectedTabId || String(before?.tab_id || '') !== expectedTabId || String(after?.tab_id || '') !== expectedTabId) return false;
    53→  if (!beforeProcess || beforeProcess !== afterProcess) return false;
    54→  if (!beforeTarget || beforeTarget !== afterTarget) return false;
    55→  return String(before?.url || '') === String(after?.url || '');
    56→}

    96→function positiveReadback(frame, fence) {
    97→  if (!fence || !isNativeFrame(frame)) return false;

   146→function scalarProjection(source, keys, max = 240) {
   147→  if (!source || typeof source !== 'object' || Array.isArray(source)) return null;
   148→  const projected = {};
   149→  for (const key of keys) {
   150→    const value = source[key];
   151→    if (value == null || typeof value === 'boolean' || typeof value === 'number') projected[key] = value ?? null;
   152→    else if (typeof value === 'string') projected[key] = clip(value, max);
   153→  }
   154→  return projected;
   155→}

   336→export async function prefetchFleetWorkerFrames({ state, executeCommand, concurrency = DEFAULT_WORKER_OBSERVATION_CONCURRENCY } = {}) {
   337→  if (typeof executeCommand !== 'function') throw new Error('supervisor_worker_prefetch_executor_required');
   338→  const limit = observationConcurrency(concurrency);
   339→  const candidates = (state?.fleet?.agents || [])
   340→    .filter((agent) => agent?.tab_id && !TERMINAL_FLEET_STATES.has(String(agent?.lifecycle_state || '').toUpperCase()))
   341→    .map((agent) => ({ agent_id: String(agent?.agent_id || ''), tab_id: String(agent.tab_id) }));
   342→  const frames = new Map();
   343→  let failures = 0;
   344→  const started = Date.now();
   345→
   346→  for (let offset = 0; offset < candidates.length; offset += limit) {
   347→    const batch = candidates.slice(offset, offset + limit);
   348→    const settled = await Promise.all(batch.map(async (agent) => {
   349→      try {
   350→        const frame = await executeCommand({ action: 'CAPTURE', payload: { tab_id: agent.tab_id }, platform: null });
   351→        return { ...agent, frame, error: null };
   352→      } catch (error) {
   353→        return { ...agent, frame: null, error: clip(error) };
   354→      }
   355→    }));
   356→    for (const row of settled) {
   357→      if (row.frame) frames.set(row.tab_id, row.frame);
   358→      else failures += 1;
   359→    }
   360→  }
   361→
   362→  return {
   363→    schema: 'metaengine.supervisor-worker-observation-prefetch.v1',
   364→    candidate_count: candidates.length,
   365→    captured_count: frames.size,
   366→    failed_count: failures,
   367→    concurrency: limit,
   368→    elapsed_ms: Math.max(0, Date.now() - started),
   369→    frames,
   370→    read_only: true,
   371→    authority_effect: false,
   372→  };
   373→}

   375→export function createSupervisorSendBoundaryExecutor({ getState, executeCommand, throttleGate = null } = {}) {
   376→  if (typeof getState !== 'function' || typeof executeCommand !== 'function') throw new Error('supervisor_send_boundary_dependencies_required');
   377→  if (throttleGate != null && (typeof throttleGate.observe !== 'function' || typeof throttleGate.active !== 'function')) {
   378→    throw new Error('supervisor_service_throttle_gate_invalid');
   379→  }
   380→  const lastNativeFrame = new Map();
   381→  const promptFence = new Map();
   382→  const readbackFence = new Map();
   383→
   384→  const rememberFrame = (tabId, frame) => {
   385→    const id = String(tabId || frame?.tab_id || '');
   386→    if (!id || !isNativeFrame(frame)) return;
   387→    lastNativeFrame.set(id, frame);
   388→    throttleGate?.observe(id, frame);
   389→    const fence = promptFence.get(id);
   390→    if (positiveReadback(frame, fence)) {
   391→      promptFence.delete(id);
   392→      readbackFence.delete(id);
   393→    }
   394→  };
   395→
   396→  const capture = async (command) => {
   397→    const tabId = String(command?.payload?.tab_id || '');
   398→    try {
   399→      const frame = await executeCommand(command);
   400→      rememberFrame(tabId, frame);
   401→      const readback = readbackFence.get(tabId);
   402→      if (readback) {
   403→        readback.remaining -= 1;
   404→        if (readback.remaining <= 0 || positiveReadback(frame, promptFence.get(tabId))) readbackFence.delete(tabId);
   405→      }
   406→      return frame;
   407→    } catch (error) {
   408→      const fence = promptFence.get(tabId);
   409→      const readback = readbackFence.get(tabId);
   410→      if (!fence || !readback) throw error;
   411→      readback.remaining -= 1;
   412→      if (readback.remaining <= 0) readbackFence.delete(tabId);
   413→      return failClosedReadbackFrame(readback.fallback || lastNativeFrame.get(tabId), tabId, error);
   414→    }
   415→  };
   416→
   417→  return async function executeWithSupervisorSendBoundary(command) {
   418→    const action = String(command?.action || '');
   419→    const tabId = String(command?.payload?.tab_id || '');
   420→
   421→    if (action === 'CAPTURE') return capture(command);
   422→
   423→    if (throttleGate?.active() && SERVICE_THROTTLE_BLOCKED_ACTIONS.has(action)) {
   424→      return suppressed(action, 'CHATGPT_SERVICE_THROTTLED', lastNativeFrame.get(tabId) || null);
   425→    }
   426→
   427→    if (action === 'SEMANTIC_TYPE') {
   428→      const baseline = lastNativeFrame.get(tabId);
   429→      if (!isNativeFrame(baseline)) return executeCommand(command);
   430→      const semanticTarget = exactSemanticTarget(baseline, command?.payload?.role, command?.payload?.accessible_name);
   431→      if (!semanticTarget?.semantic_ref && baseline.semantic_ref_context_complete !== undefined) {
   432→        return suppressed(action, 'SEMANTIC_REF_REOBSERVE_REQUIRED', baseline);
   433→      }
   434→      const text = String(command?.payload?.text ?? '');
   435→      const sha = promptSha(text);
   436→      const prior = promptFence.get(tabId);
   437→      if (prior?.prompt_sha256 === sha) {
   438→        return suppressed(action, 'DUPLICATE_PROMPT_TYPE_SUPPRESSED', baseline);
   439→      }
   440→      const fence = {
   441→        prompt_sha256: sha,
   442→        marker: promptMarker(text),
   443→        pre_url: String(baseline?.url || ''),
   444→        phase: 'TYPE_ATTEMPTED',
   445→        attempted_at: new Date().toISOString(),
   446→      };
   447→      promptFence.set(tabId, fence);
   448→      try {
   449→        const result = await executeCommand({
   450→          ...command,
   451→          payload: semanticTarget?.semantic_ref
   452→            ? { ...command.payload, semantic_ref: semanticTarget.semantic_ref }
   453→            : command.payload,
   454→        });
   455→        fence.phase = 'TYPED';
   456→        return result;
   457→      } catch (error) {
   458→        fence.phase = 'TYPE_EFFECT_AMBIGUOUS';
   459→        fence.error = clip(error?.message || error);
   460→        return suppressed(action, 'TYPE_EFFECT_AMBIGUOUS', baseline);
   461→      }
   462→    }
   463→
   464→    const isSendClick = action === 'TYPED_CLICK'
   465→      && String(command?.payload?.role || '').toLowerCase() === 'button'
   466→      && chatGptControlMatches('SEND', command?.payload?.accessible_name);
   467→    if (!isSendClick) return executeCommand(command);
   468→
   469→    const fence = promptFence.get(tabId);
   470→    if (fence?.phase === 'CLICK_ATTEMPTED') {
   471→      return suppressed(action, 'SEND_CLICK_ALREADY_ATTEMPTED', lastNativeFrame.get(tabId));
   472→    }
   473→
   474→    let before = null;
   475→    let activated = null;
   476→    const block = (reason, error = null) => {
   477→      if (fence) {
   478→        fence.phase = 'PRECLICK_BLOCKED';
   479→        if (error) fence.error = clip(error?.message || error);
   480→      }
   481→      const fallback = activated || before || lastNativeFrame.get(tabId) || null;
   482→      readbackFence.set(tabId, { remaining: 6, fallback });
   483→      return suppressed(action, reason, fallback);
   484→    };
   485→
   486→    try {
   487→      before = await executeCommand({ action: 'CAPTURE', payload: { tab_id: tabId }, platform: command?.platform ?? null });
   488→      rememberFrame(tabId, before);
   489→      // Non-native executors are test/model adapters and retain the proven core path.
   490→      // The installed Electron path always exposes metaengine.native-browser.perception.v1.
   491→      if (!isNativeFrame(before)) return executeCommand(command);
   492→      if (throttleGate?.active()) return block('CHATGPT_SERVICE_THROTTLED');
   493→
   494→      await executeCommand({ action: 'SELECT_TAB', payload: { tab_id: tabId }, platform: command?.platform ?? null });
   495→      const state = await getState();
   496→      if (!exactSelectedTab(state, tabId)) return block('SUPERVISOR_TAB_NOT_EXACTLY_SELECTED');
   497→
   498→      activated = await executeCommand({ action: 'CAPTURE', payload: { tab_id: tabId }, platform: command?.platform ?? null });
   499→      rememberFrame(tabId, activated);
   500→      if (!isNativeFrame(activated)) return block('SUPERVISOR_NATIVE_FRAME_LOST');
   501→      if (throttleGate?.active()) return block('CHATGPT_SERVICE_THROTTLED');
   502→      if (!exactIncarnation(before, activated, tabId)) return block('SUPERVISOR_TARGET_INCARNATION_CHANGED');
   503→      if (!positiveViewport(activated)) return block('SUPERVISOR_VIEWPORT_NOT_VISIBLE');
   504→
   505→      const send = uniqueChatGptControl(activated, 'SEND');
   506→      if (!send) return block('SUPERVISOR_SEND_NOT_UNIQUE_AFTER_SELECT');
   507→      if (String(send.name || '') !== String(command?.payload?.accessible_name || '')) return block('SUPERVISOR_SEND_CONTROL_CHANGED');
   508→      if (!send.semantic_ref && activated.semantic_ref_context_complete !== undefined) {
   509→        return block('SEMANTIC_REF_REOBSERVE_REQUIRED');
   510→      }
   511→
   512→      if (fence) fence.phase = 'CLICK_ATTEMPTED';
   513→      readbackFence.set(tabId, { remaining: 6, fallback: activated });
   514→      try {
   515→        return await executeCommand({
   516→          ...command,
   517→          payload: send.semantic_ref
   518→            ? { ...command.payload, semantic_ref: send.semantic_ref }
   519→            : command.payload,
   520→        });
   521→      } catch (error) {
   522→        if (fence) fence.error = clip(error?.message || error);
   523→        return suppressed(action, 'SEND_CLICK_EFFECT_AMBIGUOUS', activated);
   524→      }
   525→    } catch (error) {
   526→      return block('SUPERVISOR_SEND_BOUNDARY_REVALIDATION_FAILED', error);
   527→    }
   528→  };
   529→}
   530→
   531→// Static continuity markers retained because the implementation itself remains in
   532→// supervisor-lifecycle-runtime-core.mjs: CHATGPT_CONVERSATION_LIMIT_HINT and
   533→// reason.startsWith('MAX_CYCLES_PER_EPOCH'). The wrapper adds no page authority.
   534→export class SupervisorLifecycleRuntime extends CoreSupervisorLifecycleRuntime {
   535→  #serviceThrottleGate;
   536→  #sourceGetState;
   537→  #sourceExecute;
   538→  #workerCaptureCache;
   539→  #workerObservationConcurrency;
   540→  #lastWorkerPrefetch = null;
   541→
   542→  constructor(options = {}) {
   543→    const serviceThrottleGate = options.serviceThrottleGate || new ChatGptServiceThrottleGate();
   544→    if (typeof serviceThrottleGate.active !== 'function' || typeof serviceThrottleGate.snapshot !== 'function') {
   545→      throw new Error('supervisor_service_throttle_gate_invalid');
   546→    }
   547→    if (typeof options.getState !== 'function' || typeof options.executeCommand !== 'function') {
   548→      throw new Error('supervisor_lifecycle_dependencies_required');
   549→    }
   550→    const baseCanActuate = typeof options.canActuate === 'function' ? options.canActuate : () => true;
   551→    const workerCaptureCache = new Map();
   552→    const sourceExecute = options.executeCommand;
   553→    const cachedExecute = async (command) => {
   554→      if (String(command?.action || '') === 'CAPTURE') {
   555→        const tabId = String(command?.payload?.tab_id || '');
   556→        if (tabId && workerCaptureCache.has(tabId)) {
   557→          const frame = workerCaptureCache.get(tabId);
   558→          workerCaptureCache.delete(tabId);
   559→          return frame;
   560→        }
   561→      }
   562→      return sourceExecute(command);
   563→    };
   564→    const guardedExecute = createSupervisorSendBoundaryExecutor({
   565→      getState: options.getState,
   566→      executeCommand: cachedExecute,
   567→      throttleGate: serviceThrottleGate,
   568→    });
   569→    super({
   570→      ...options,
   571→      canActuate: () => baseCanActuate() === true && serviceThrottleGate.active() !== true,
   572→      executeCommand: guardedExecute,
   573→    });
   574→    this.#serviceThrottleGate = serviceThrottleGate;
   575→    this.#sourceGetState = options.getState;
   576→    this.#sourceExecute = sourceExecute;
   577→    this.#workerCaptureCache = workerCaptureCache;
   578→    this.#workerObservationConcurrency = observationConcurrency(options.workerObservationConcurrency ?? options.workerObservationBudget);
   579→  }
   580→
   581→  async cycle(options = {}) {
   582→    this.#workerCaptureCache.clear();
   583→    try {
   584→      const state = await this.#sourceGetState();
   585→      const prefetched = await prefetchFleetWorkerFrames({
   586→        state,
   587→        executeCommand: this.#sourceExecute,
   588→        concurrency: this.#workerObservationConcurrency,
   589→      });
   590→      for (const [tabId, frame] of prefetched.frames.entries()) this.#workerCaptureCache.set(tabId, frame);
   591→      this.#lastWorkerPrefetch = Object.freeze({
   592→        schema: prefetched.schema,
   593→        candidate_count: prefetched.candidate_count,
   594→        captured_count: prefetched.captured_count,
   595→        failed_count: prefetched.failed_count,
   596→        concurrency: prefetched.concurrency,
   597→        elapsed_ms: prefetched.elapsed_ms,
   598→        read_only: true,
   599→        authority_effect: false,
   600→      });
   601→    } catch (error) {
   602→      this.#lastWorkerPrefetch = Object.freeze({
   603→        schema: 'metaengine.supervisor-worker-observation-prefetch.v1',
   604→        candidate_count: null,
   605→        captured_count: 0,
   606→        failed_count: null,
   607→        concurrency: this.#workerObservationConcurrency,
   608→        elapsed_ms: null,
   609→        error: clip(error),
   610→        read_only: true,
   611→        authority_effect: false,
   612→      });
   613→    }
   614→    try {
   615→      return await super.cycle(options);
   616→    } finally {
   617→      this.#workerCaptureCache.clear();
   618→    }
   619→  }
   620→
   621→  snapshot() {
   622→    const base = super.snapshot();
   623→    return {
   624→      ...base,
   625→      continuous_service: {
   626→        ...(base.continuous_service || {}),
   627→        service_throttle_backpressure: 'UI_READBACK_GATE_V1',
   628→        worker_observation_prefetch: 'BOUNDED_PARALLEL_READ_ONLY_V1',
   629→        worker_observation_concurrency: this.#workerObservationConcurrency,
   630→      },
   631→      worker_observation_prefetch: this.#lastWorkerPrefetch ? structuredClone(this.#lastWorkerPrefetch) : null,
   632→      service_throttle: this.#serviceThrottleGate.snapshot(),
   633→      authority_effect: false,
   634→    };
   635→  }
   636→
   637→  statusSnapshot() {
   638→    return buildSupervisorLifecycleStatusSnapshot(this.snapshot());
   639→  }
   640→}
