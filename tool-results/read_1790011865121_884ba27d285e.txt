    37→    37→function sanitizeActiveWake(input) {

   115→   115→function sanitize(input) {
   116→   116→  const base = freshState();
   117→   117→  if (!input || input.schema !== base.schema || input.supervisor_id !== SUPERVISOR_ID) return base;
   118→   118→  let conversationUrl = null;
   119→   119→  try { if (input.conversation_url) conversationUrl = normalizeUrl(input.conversation_url); } catch {}
   120→   120→  const queued = Array.isArray(input.queued_wakes)
   121→   121→    ? input.queued_wakes.filter((row) => row && WAKE_REASONS.has(String(row.reason))).slice(-MAX_QUEUED_WAKES).map(clone)
   122→   122→    : [];
   123→   123→  const ambiguousHistory = Array.isArray(input.ambiguous_history)
   124→   124→    ? input.ambiguous_history.filter((row) => row && typeof row === 'object').slice(-MAX_WAKE_HISTORY).map(clone)
   125→   125→    : [];
   126→   126→  const predecessorWakeHistory = Array.isArray(input.predecessor_wake_history)
   127→   127→    ? input.predecessor_wake_history.filter((row) => row && typeof row === 'object').slice(-MAX_WAKE_HISTORY).map(clone)
   128→   128→    : [];
   129→   129→  return {
   130→   130→    ...base,
   131→   131→    supervisor_epoch: Math.max(1, Number(input.supervisor_epoch) || 1),
   132→   132→    cycle_seq: Math.max(0, Number(input.cycle_seq) || 0),
   133→   133→    state: input.paused === true ? 'PAUSED' : (KEEPALIVE_STATES.includes(input.state) ? input.state : 'RECOVERING'),
   134→   134→    conversation_url: conversationUrl,
   135→   135→    tab_id: input.tab_id ? String(input.tab_id) : null,
   136→   136→    paused: input.paused === true,
   137→   137→    paused_from_state: KEEPALIVE_STATES.includes(input.paused_from_state) ? input.paused_from_state : null,
   138→   138→    process_incarnation_id: sanitizeProcessIncarnationId(input.process_incarnation_id),
   139→   139→    process_incarnation_started_at: input.process_incarnation_started_at || null,
   140→   140→    predecessor_process_incarnation_id: sanitizeProcessIncarnationId(input.predecessor_process_incarnation_id),
   141→   141→    predecessor_fenced_at: input.predecessor_fenced_at || null,
   142→   142→    predecessor_queued_wake_count: Math.max(0, Number(input.predecessor_queued_wake_count) || 0),
   143→   143→    admission_state: ADMISSION_STATES.has(String(input.admission_state || '').toUpperCase())
   144→   144→      ? String(input.admission_state).toUpperCase()
   145→   145→      : 'UNKNOWN',
   146→   146→    admission_reason: input.admission_reason ? String(input.admission_reason).slice(0, 160) : null,
   147→   147→    admission_generation_floor: typeof input.admission_generation_floor === 'number'
   148→   148→      && Number.isSafeInteger(input.admission_generation_floor)
   149→   149→      ? Math.max(0, input.admission_generation_floor)
   150→   150→      : null,
   151→   151→    admission_refill_enabled: typeof input.admission_refill_enabled === 'boolean' ? input.admission_refill_enabled : null,
   152→   152→    admission_supervisor_enabled: typeof input.admission_supervisor_enabled === 'boolean' ? input.admission_supervisor_enabled : null,
   153→   153→    admission_observed_at: input.admission_observed_at || null,
   154→   154→    parked_at: input.parked_at || null,
   155→   155→    parked_reason: input.parked_reason ? String(input.parked_reason).slice(0, 160) : null,
   156→   156→    parked_queued_wake_count: Math.max(0, Number(input.parked_queued_wake_count) || 0),
   157→   157→    parked_wake_reasons: Array.isArray(input.parked_wake_reasons)
   158→   158→      ? input.parked_wake_reasons.map((value) => String(value).slice(0, 80)).slice(-MAX_QUEUED_WAKES)
   159→   159→      : [],
   160→   160→    suppressed_wake_count: Math.max(0, Number(input.suppressed_wake_count) || 0),
   161→   161→    last_suppressed_wake_at: input.last_suppressed_wake_at || null,
   162→   162→    last_suppressed_wake_reason: input.last_suppressed_wake_reason ? String(input.last_suppressed_wake_reason).slice(0, 80) : null,
   163→   163→    queued_wakes: queued,
   164→   164→    pending_wake: input.pending_wake && typeof input.pending_wake === 'object' ? clone(input.pending_wake) : null,
   165→   165→    active_wake: sanitizeActiveWake(input.active_wake),
   166→   166→    predecessor_wake_history: predecessorWakeHistory,
   167→   167→    ambiguous_history: ambiguousHistory,
   168→   168→    last_wake_at: input.last_wake_at || null,
   169→   169→    last_wake_reason: input.last_wake_reason || null,
   170→   170→    last_unsent_attempt_at: input.last_unsent_attempt_at || null,
   171→   171→    last_completed_cycle_at: input.last_completed_cycle_at || null,
   172→   172→    last_research_wake_at: input.last_research_wake_at || null,
   173→   173→    previous_worker_generation: sanitizeWorkerGenerationMap(input.previous_worker_generation),
   174→   174→    rollover_reason: input.rollover_reason ? String(input.rollover_reason).slice(0, 160) : null,
   175→   175→    rollover_release_at: input.rollover_release_at || null,
   176→   176→    rollover_attempt: input.rollover_attempt && typeof input.rollover_attempt === 'object' ? clone(input.rollover_attempt) : null,
   177→   177→    updated_at: input.updated_at || null,
   178→   178→  };
   179→   179→}

   222→   222→export class SupervisorKeepalive {
   223→   223→  #load; #save; #clock; #uuid; #state; #minWakeIntervalMs; #maxCyclesPerEpoch; #processIncarnationId;
   224→   224→
   225→   225→  constructor({ loadState, saveState, clock = () => Date.now(), uuid = () => crypto.randomUUID(), processIncarnationId = null, minWakeIntervalMs = 60000, maxCyclesPerEpoch = null } = {}) {
   226→   226→    if (typeof loadState !== 'function' || typeof saveState !== 'function') throw new Error('keepalive_persistence_required');
   227→   227→    this.#load = loadState;
   228→   228→    this.#save = saveState;
   229→   229→    this.#clock = clock;
   230→   230→    this.#uuid = uuid;
   231→   231→    this.#processIncarnationId = sanitizeProcessIncarnationId(processIncarnationId)
   232→   232→      || `process_${crypto.randomUUID().toLowerCase()}`;
   233→   233→    this.#minWakeIntervalMs = Math.max(30000, Number(minWakeIntervalMs) || 60000);
   234→   234→    void maxCyclesPerEpoch;
   235→   235→    this.#maxCyclesPerEpoch = null;
   236→   236→    this.#state = freshState();
   237→   237→  }
   238→   238→
   239→   239→  async init() {
   240→   240→    this.#state = sanitize(await this.#load());
   241→   241→    const recoveredAt = iso(this.#clock);
   242→   242→    const predecessorIncarnation = this.#state.process_incarnation_id;
   243→   243→    const crossedProcessBoundary = predecessorIncarnation !== this.#processIncarnationId;
   244→   244→
   245→   245→    if (crossedProcessBoundary) {
   246→   246→      const hasPredecessor = predecessorIncarnation != null;
   247→   247→      const pendingProcess = sanitizeProcessIncarnationId(this.#state.pending_wake?.process_incarnation_id);
   248→   248→      if (hasPredecessor
   249→   249→        && this.#state.pending_wake?.ambiguous_at
   250→   250→        && this.#state.pending_wake?.automatic_retry_allowed === false
   251→   251→        && pendingProcess
   252→   252→        && pendingProcess !== this.#processIncarnationId
   253→   253→        && !this.#state.pending_wake.process_boundary_fenced_at) {
   254→   254→        // Persist the safety fact on the ambiguous wake itself. The immediate
   255→   255→        // predecessor pointer is intentionally one-hop and is overwritten on
   256→   256→        // every restart; this wake-local fence survives any number of later
   257→   257→        // process boundaries without ever making the old effect retryable.
   258→   258→        this.#state.pending_wake.process_boundary_fenced_at = recoveredAt;
   259→   259→        this.#state.pending_wake.process_boundary_fenced_by_process_incarnation_id = predecessorIncarnation;
   260→   260→      }
   261→   261→      const predecessorActive = hasPredecessor ? this.#state.active_wake : null;
   262→   262→      if (predecessorActive) {
   263→   263→        this.#state.predecessor_wake_history = [
   264→   264→          ...this.#state.predecessor_wake_history,
   265→   265→          {
   266→   266→            ...clone(predecessorActive),
   267→   267→            retired_at: recoveredAt,
   268→   268→            retired_reason: 'PROCESS_BOUNDARY_ACTIVE_WAKE_FENCED',
   269→   269→            automatic_retry_allowed: false,
   270→   270→            authority_effect: false,
   271→   271→          },
   272→   272→        ].slice(-MAX_WAKE_HISTORY);
   273→   273→      }
   274→   274→      if (hasPredecessor) {
   275→   275→        this.#state.predecessor_process_incarnation_id = predecessorIncarnation;
   276→   276→        this.#state.predecessor_fenced_at = recoveredAt;
   277→   277→        this.#state.predecessor_queued_wake_count = this.#state.queued_wakes.length;
   278→   278→      } else {
   279→   279→        this.#state.predecessor_process_incarnation_id = null;
   280→   280→        this.#state.predecessor_fenced_at = null;
   281→   281→        this.#state.predecessor_queued_wake_count = 0;
   282→   282→      }
   283→   283→      this.#state.queued_wakes = [];
   284→   284→      this.#state.active_wake = null;
   285→   285→      this.#state.previous_worker_generation = {};
   286→   286→      this.#state.process_incarnation_id = this.#processIncarnationId;
   287→   287→      this.#state.process_incarnation_started_at = recoveredAt;
   288→   288→    }
   289→   289→
   290→   290→    if (this.#state.pending_wake && !this.#state.pending_wake.ambiguous_at) {
   291→   291→      this.#state.pending_wake.ambiguous_at = recoveredAt;
   292→   292→      this.#state.pending_wake.ambiguous_reason = 'PROCESS_RESTART_WITH_UNRESOLVED_WAKE_EFFECT';
   293→   293→      this.#state.pending_wake.restart_recovered_at = recoveredAt;
   294→   294→      this.#state.pending_wake.automatic_retry_allowed = false;
   295→   295→    }
   296→   296→    if (this.#state.state === 'ROLLOVER_PENDING') {
   297→   297→      this.#state.state = 'ROLLOVER_AMBIGUOUS';
   298→   298→      this.#state.rollover_reason = 'PROCESS_RESTART_WITH_UNRESOLVED_ROLLOVER_EFFECT';
   299→   299→      this.#state.rollover_attempt = {
   300→   300→        ...(this.#state.rollover_attempt || {}),
   301→   301→        restart_recovered_at: recoveredAt,
   302→   302→        automatic_retry_allowed: false,
   303→   303→      };
   304→   304→    }
   305→   305→    if (crossedProcessBoundary && this.#state.admission_state === 'OPEN') {
   306→   306→      // An OPEN decision is never trusted across a process boundary. A fresh
   307→   307→      // authoritative heartbeat must reopen continuous-service actuation.
   308→   308→      this.#state.admission_state = 'UNKNOWN';
   309→   309→      this.#state.admission_reason = 'PROCESS_RESTART_REOBSERVATION_REQUIRED';
   310→   310→    }
   311→   311→    if (this.#state.paused) this.#state.state = 'PAUSED';
   312→   312→    else if (this.#state.pending_wake) this.#state.state = 'WAKE_AMBIGUOUS';
   313→   313→    else if (['ROLLOVER_DEFERRED','ROLLOVER_REQUIRED','ROLLOVER_AMBIGUOUS'].includes(this.#state.state)) {}
   314→   314→    else if (this.#state.active_wake) this.#state.state = 'ACTIVE';
   315→   315→    else if (this.#state.admission_state === 'CLOSED') this.#state.state = 'PARKED';
   316→   316→    else this.#state.state = this.#state.conversation_url ? 'WAITING' : 'RECOVERING';
   317→   317→    await this.#persist();
   318→   318→    return this.snapshot();
   319→   319→  }
   320→   320→
   321→   321→  snapshot() {
   322→   322→    return Object.freeze({
   323→   323→      ...clone(this.#state),
   324→   324→      work_cycle_limit: this.#maxCyclesPerEpoch,
   325→   325→      worker_generation_memory_limit: MAX_WORKER_GENERATION_MEMORY,
   326→   326→      automatic_rollover_cycle_limit_enabled: false,
   327→   327→      external_confirmation_required_for_continuation: false,
   328→   328→    });
   329→   329→  }
   330→   330→  activeWake() { return clone(this.#state.active_wake); }
   331→   331→
   332→   332→  async applyAdmissionClosed(control = {}) {
   333→   333→    if (control?.authoritative !== true
   334→   334→      || control?.state !== 'CLOSED'
   335→   335→      || control?.continuous_service_allowed !== false
   336→   336→      || typeof control?.refill_enabled !== 'boolean'
   337→   337→      || typeof control?.supervisor_admission_enabled !== 'boolean'
   338→   338→      || typeof control?.generation_floor !== 'number'
   339→   339→      || !Number.isSafeInteger(control.generation_floor)) {
   340→   340→      throw new Error('keepalive_admission_closed_readback_invalid');
   341→   341→    }
   342→   342→    if (this.#state.admission_generation_floor != null
   343→   343→      && control.generation_floor < this.#state.admission_generation_floor) {
   344→   344→      throw new Error('keepalive_admission_generation_floor_regression');
   345→   345→    }
   346→   346→    const observedAt = iso(this.#clock);
   347→   347→    this.#state.admission_state = 'CLOSED';
   348→   348→    this.#state.admission_reason = String(control.reason || 'CONTINUOUS_SERVICE_ADMISSION_FENCED').slice(0, 160);

   653→   653→    for (const reason of ['WORKER_LOST','WORKER_FAILED']) {
   654→   654→      const agentIds = events.filter((row) => row.reason === reason).map((row) => row.agent_id);
   655→   655→      if (agentIds.length) await this.enqueueWake(reason, { key: 'fleet-terminal-burst', agent_ids: agentIds });
   656→   656→    }

   660→   660→  canWake() {
   661→   661→    if (this.#state.admission_state === 'CLOSED'
   662→   662→      || this.#state.paused
   663→   663→      || ['PARKED','WAKE_AMBIGUOUS','ROLLOVER_DEFERRED','ROLLOVER_REQUIRED','ROLLOVER_PENDING','ROLLOVER_AMBIGUOUS'].includes(this.#state.state)) return false;
   664→   664→    const actionable = this.#state.queued_wakes.filter((row) => row.process_incarnation_id === this.#processIncarnationId);
   665→   665→    if (!this.#state.conversation_url || this.#state.pending_wake || this.#state.active_wake || actionable.length === 0) return false;
   666→   666→    // D-K5 (live 2026-09-19): a wake that was prepared but provably NOT sent
   667→   667→    // (pre-effect throw: composer resolution, replace unverified, stale ref)
   668→   668→    // used to be re-prepared on the very next monitor tick — and the
   669→   669→    // CONTINUE_DEVELOPMENT reason bypasses the wake interval entirely, so a
   670→   670→    // permanent pre-effect failure became a ~2s retry storm that raced the
   671→   671→    // page's draft-sync and appended ~3k chars of wake text per MINUTE to the
   672→   672→    // composer. One unsent attempt must cost at least the wake interval.
   673→   673→    if (this.#state.last_unsent_attempt_at) {
   674→   674→      const since = this.#clock() - new Date(this.#state.last_unsent_attempt_at).getTime();
   675→   675→      if (Number.isFinite(since) && since < this.#minWakeIntervalMs) return false;
   676→   676→    }
   677→   677→    if (!this.#state.last_wake_at) return true;
   678→   678→    if (String(actionable[0]?.reason || '') === 'CONTINUE_DEVELOPMENT') return true;
   679→   679→    return this.#clock() - new Date(this.#state.last_wake_at).getTime() >= this.#minWakeIntervalMs;
   680→   680→  }

   682→   682→  async prepareNextWake() {
   683→   683→    if (!this.canWake()) return { ok: false, suppressed: true, state: this.#state.state, authority_effect: false };
   684→   684→    const queued = this.#state.queued_wakes.find((row) => row.process_incarnation_id === this.#processIncarnationId);
   685→   685→    const wakeId = `wake_${String(this.#uuid()).replace(/[^a-z0-9-]/gi, '').toLowerCase()}`;
   686→   686→    const pending = {
   687→   687→      wake_id: wakeId,
   688→   688→      reason: queued.reason,
   689→   689→      queue_key: queued.key,
   690→   690→      prepared_at: iso(this.#clock),
   691→   691→      supervisor_epoch: this.#state.supervisor_epoch,
   692→   692→      cycle_seq: this.#state.cycle_seq + 1,
   693→   693→      process_incarnation_id: this.#processIncarnationId,
   694→   694→      automatic_retry_allowed: false,
   695→   695→    };
   696→   696→    this.#state.pending_wake = pending;
   697→   697→    this.#state.state = 'WAKE_PENDING';
   698→   698→    await this.#persist();
   699→   699→    return {
   700→   700→      ok: true,
   701→   701→      pending: clone(pending),
   702→   702→      conversation_url: this.#state.conversation_url,
   703→   703→      tab_id: this.#state.tab_id,
   704→   704→      message: buildSupervisorWakeMessage({ supervisorEpoch: pending.supervisor_epoch, cycleSeq: pending.cycle_seq, wakeId, reason: pending.reason }),
   705→   705→      authority_effect: false,
   706→   706→    };
   707→   707→  }

   714→   714→  async confirmWakeSent(wakeId) {
   715→   715→    const pending = this.#state.pending_wake;
   716→   716→    if (!pending || pending.wake_id !== String(wakeId)) throw new Error('keepalive_wake_binding_mismatch');
   717→   717→    this.#state.cycle_seq = pending.cycle_seq;
   718→   718→    this.#state.last_wake_at = iso(this.#clock);
   719→   719→    this.#state.last_wake_reason = pending.reason;
   720→   720→    this.#state.last_unsent_attempt_at = null;
   721→   721→    this.#state.queued_wakes = this.#state.queued_wakes.filter((row) => row.key !== pending.queue_key);
   722→   722→    this.#state.pending_wake = null;
   723→   723→    this.#state.active_wake = {
   724→   724→      ...clone(pending),
   725→   725→      origin_process_incarnation_id: sanitizeProcessIncarnationId(pending.process_incarnation_id),
   726→   726→      process_incarnation_id: this.#processIncarnationId,
   727→   727→      confirmed_at: iso(this.#clock),
   728→   728→    };
   729→   729→    delete this.#state.active_wake.automatic_retry_allowed;
   730→   730→    this.#state.state = 'ACTIVE';
   731→   731→    await this.#persist();

   733→   733→  }

   746→   746→    const pending = this.#state.pending_wake;
   747→   747→    if (!pending || pending.wake_id !== String(wakeId)) throw new Error('keepalive_wake_binding_mismatch');
   748→   748→    pending.ambiguous_at = iso(this.#clock);

   750→   750→    pending.automatic_retry_allowed = false;

   752→   752→    // remember the tab it typed into. Without it the recovery scoping falls
   753→   753→    // back to keepalive.tab_id, which still points at a tab from a previous

   759→   759→    }
   760→   760→    this.#state.state = 'WAKE_AMBIGUOUS';
   761→   761→    await this.#persist();

   766→   766→    const pending = this.#state.pending_wake;

   769→   769→    if (pending.ambiguity_continuation_attempted_at) return false;

   774→   774→    pending.ambiguity_continuation_tab_id = tabId;
   775→   775→    pending.ambiguity_continuation_composer_sha256 = composerSha;
   776→   776→    pending.automatic_retry_allowed = false;

   781→   781→  async retireAmbiguousAfterProcessBoundary({ reason = 'PROCESS_BOUNDARY_ORIGINAL_BOOTSTRAP_TARGET_LOST', replacement_tab_id = null } = {}) {
   782→   782→    const pending = this.#state.pending_wake;
   783→   783→    if (!pending || !pending.ambiguous_at) throw new Error('keepalive_no_ambiguous_wake');
   784→   784→    const pendingProcess = sanitizeProcessIncarnationId(pending.process_incarnation_id);
   785→   785→    const currentProcess = sanitizeProcessIncarnationId(this.#state.process_incarnation_id);
   786→   786→    const predecessorProcess = sanitizeProcessIncarnationId(this.#state.predecessor_process_incarnation_id);
   787→   787→    if (!pendingProcess || !currentProcess || pendingProcess === currentProcess) {
   788→   788→      throw new Error('keepalive_process_boundary_ambiguity_not_proven');
   789→   789→    }
   790→   790→    const immediatePredecessorFence = predecessorProcess === pendingProcess
   791→   791→      && Boolean(this.#state.predecessor_fenced_at);
   792→   792→    const durableWakeBoundaryFence = Boolean(pending.process_boundary_fenced_at)
   793→   793→      && pending.automatic_retry_allowed === false;
   794→   794→    if (!this.#state.predecessor_fenced_at || (!immediatePredecessorFence && !durableWakeBoundaryFence)) {
   795→   795→      throw new Error('keepalive_process_boundary_predecessor_not_proven');
   796→   796→    }
   797→   797→    const retiredAt = iso(this.#clock);
   798→   798→    this.#state.cycle_seq = Math.max(this.#state.cycle_seq, Math.max(1, Number(pending.cycle_seq) || 1));
   799→   799→    this.#state.queued_wakes = this.#state.queued_wakes.filter((row) => !(
   800→   800→      row.key === pending.queue_key
   801→   801→      && sanitizeProcessIncarnationId(row.process_incarnation_id) === pendingProcess
   802→   802→    ));
   803→   803→    this.#state.ambiguous_history = [
   804→   804→      ...this.#state.ambiguous_history,
   805→   805→      {
   806→   806→        ...clone(pending),
   807→   807→        retired_at: retiredAt,
   808→   808→        retired_reason: String(reason || 'PROCESS_BOUNDARY_ORIGINAL_BOOTSTRAP_TARGET_LOST').slice(0, 200),
   809→   809→        retired_process_incarnation_id: pendingProcess,
   810→   810→        recovery_process_incarnation_id: currentProcess,
   811→   811→        replacement_tab_id: replacement_tab_id ? String(replacement_tab_id).slice(0, 120) : null,
   812→   812→        automatic_retry_allowed: false,
   813→   813→      },
   814→   814→    ].slice(-MAX_WAKE_HISTORY);
   815→   815→    this.#state.pending_wake = null;
   816→   816→    this.#state.last_completed_cycle_at = retiredAt;
   817→   817→    this.#state.state = this.#postWakeSettlementState('RECOVERING');
   818→   818→    await this.#persist();
   819→   819→    return this.snapshot();
   820→   820→  }

   848→   848→  async resolveAmbiguous({ observed_sent } = {}) {
   849→   849→    if (this.#state.state !== 'WAKE_AMBIGUOUS' || !this.#state.pending_wake) throw new Error('keepalive_no_ambiguous_wake');
   850→   850→    if (observed_sent === true) return this.confirmWakeSent(this.#state.pending_wake.wake_id);
   851→   851→    if (observed_sent === false) {
   852→   852→      // D-K5: the wake was provably not sent — record the attempt so the
   853→   853→      // retry pays the wake-interval price instead of storming.
   854→   854→      this.#state.last_unsent_attempt_at = iso(this.#clock);
   855→   855→      this.#state.pending_wake = null;
   856→   856→      this.#state.state = this.#postWakeSettlementState();
   857→   857→      await this.#persist();
   858→   858→      return this.snapshot();
   859→   859→    }
   860→   860→    throw new Error('keepalive_ambiguous_resolution_requires_observation');
   861→   861→  }
   862→   862→
   863→   863→  // LIVE 2026-09-21 (self-update restart window): a wake that went
   864→   864→  // TYPE_EFFECT_AMBIGUOUS exactly across the successor boot can fence the
   865→   865→  // rollover FOREVER — the lifecycle's rollover retire paths only accept a
   866→   866→  // pending wake in WAKE_AMBIGUOUS, so ROLLOVER_REQUIRED + pending-ambiguous
   867→   867→  // deadlocks the lifecycle (keepalive_no_ambiguous_wake on every tick) and
   868→   868→  // the idle maintenance window never idles, which starves the devos task
   869→   869→  // cycle (no leases). This settlement is proof-based, from ROLLOVER_REQUIRED:
   870→   870→  //   observed_sent === true  → the marker is provably in the transcript:
   871→   871→  //                             confirm the wake (resume normal processing).
   872→   872→  //   observed_sent === false → the wake provably did not cross (composer
   873→   873→  //                             still holds the exact message / terminal and
   874→   874→  //                             no marker): drop it; postWakeSettlementState
   875→   875→  //                             PRESERVES the ROLLOVER_* state so the
   876→   876→  //                             rollover proceeds on a clean ledger, and the
   877→   877→  //                             wake re-prepares later at the wake-interval
   878→   878→  //                             price (D-K5 anti-storm).
   879→   879→  async settleRolloverBlockedAmbiguousWake({ observed_sent } = {}) {
   880→   880→    const pending = this.#state.pending_wake;
   881→   881→    if (this.#state.state !== 'ROLLOVER_REQUIRED' || !pending?.ambiguous_at) throw new Error('keepalive_no_rollover_blocked_ambiguous_wake');
   882→   882→    if (observed_sent === true) return this.confirmWakeSent(pending.wake_id);
   883→   883→    if (observed_sent === false) {
   884→   884→      this.#state.last_unsent_attempt_at = iso(this.#clock);
   885→   885→      this.#state.pending_wake = null;
   886→   886→      this.#state.state = this.#postWakeSettlementState();
   887→   887→      await this.#persist();
   888→   888→      return this.snapshot();
   889→   889→    }
   890→   890→    throw new Error('keepalive_ambiguous_resolution_requires_observation');
   891→   891→  }
